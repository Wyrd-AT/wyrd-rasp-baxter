#define MAX_LOCKED_ASSETS 10

struct TargetAsset {
    String mac = "";
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInercia = 0;
    int mediaRSSI = -100;
    bool emDisputa = false; // NOVO: Flag para saber se já enviámos um GET
    unsigned long inicioDisputa = 0;
    uint32_t idTransacao = 0; // NOVO: ID de transação para esta disputa
};

struct LockedAsset {
    String mac = "";
    bool confirmada = false;
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInerciaSaida = 0;
} lockedAsset;

LockedAsset lockedAssets[MAX_LOCKED_ASSETS]; // <-- NOVO: Array para os ativos presentes
int lockedAssetsCount = 0;    

#define MAX_AVAILABLE_ASSETS 50
TargetAsset potentialTargets[MAX_AVAILABLE_ASSETS];
int potentialTargetsCount = 0;

void lerConfiguracoes() {

    String saved_server_ip = preferences.getString("server_ip", "10.0.0.124");
    String saved_server_port = preferences.getString("server_port", "8080");

    String saved_mqtt_port = preferences.getString("mqtt_port", "1883");
    strcpy(SERVER_IP, saved_server_ip.c_str());
    strcpy(SERVER_PORT_STR, saved_server_port.c_str());
    strcpy(MQTT_BROKER_HOST, saved_mqtt_host.c_str());
    strcpy(MQTT_BROKER_PORT_STR, saved_mqtt_port.c_str());
    SERVER_PORT = atoi(SERVER_PORT_STR);
    MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
    preferences.end();
}

void salvarConfiguracoes() {
    preferences.begin("rtls-config", false);
    preferences.putString("server_ip", SERVER_IP);
    preferences.putString("server_port", SERVER_PORT_STR);
    preferences.putString("mqtt_host", MQTT_BROKER_HOST);
    preferences.putString("mqtt_port", MQTT_BROKER_PORT_STR);
    preferences.end();
}

void saveConfigCallback () {
    shouldSaveConfig = true;
}

// =================== LÓGICA DE COMUNICAÇÃO ===================
class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) override {
        String foundMac = advertisedDevice.getAddress().toString().c_str();

        // --- MUDANÇA AQUI: Verifica a lista de ativos travados ---
        for (int i = 0; i < lockedAssetsCount; i++) {
            if (foundMac.equalsIgnoreCase(lockedAssets[i].mac)) {
                lockedAssets[i].rssiHistory[lockedAssets[i].index] = advertisedDevice.getRSSI();
                lockedAssets[i].index = (lockedAssets[i].index + 1) % RSSI_HISTORY_SIZE;
                if (lockedAssets[i].index == 0) lockedAssets[i].isFull = true;
                return; // Encontrou o ativo, pode parar a busca
            }
        }

        for (int i = 0; i < potentialTargetsCount; i++) {
            if (foundMac.equalsIgnoreCase(potentialTargets[i].mac)) {
                TargetAsset &target = potentialTargets[i];
                target.rssiHistory[target.index] = advertisedDevice.getRSSI();
                target.index = (target.index + 1) % RSSI_HISTORY_SIZE;
                if (target.index == 0) target.isFull = true;
                break;
            }
        }
    }
};

void mqtt_callback(char* topic, byte* payload, unsigned int length) {
    Serial.println("------------------------------------");
    Serial.printf("Mensagem recebida no tópico: %s\n", topic);
    String topicStr = String(topic);

    // --- LÓGICA DE VEREDITO (Atualizada para multi-ativo) ---
    if (topicStr == MQTT_VERDICT_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        uint32_t idTransacaoRecebida = doc["transacao_id"];
        String ativo_mac = doc["ativo"];

        // Procura pelo alvo que corresponde a ESTE ATIVO e a ESTA TRANSAÇÃO
        for (int i = 0; i < potentialTargetsCount; i++) {
            
            // --- CONDIÇÃO CORRIGIDA E MAIS SEGURA ---
            if (potentialTargets[i].emDisputa && 
                potentialTargets[i].mac.equalsIgnoreCase(ativo_mac) &&
                potentialTargets[i].idTransacao == idTransacaoRecebida) {
                // --- FIM DA CORREÇÃO ---

                const char* status = doc["status"];
                
                if (strcmp(status, "WIN") == 0 && lockedAssetsCount < MAX_LOCKED_ASSETS) {
                    Serial.printf(">>> VEREDITO VÁLIDO (ID %u): VENCEU! Travando no ativo %s\n", idTransacaoRecebida, ativo_mac.c_str());
                    // Adiciona o novo ativo à lista de presentes
                    lockedAssets[lockedAssetsCount] = {};
                    lockedAssets[lockedAssetsCount].mac = ativo_mac;
                    lockedAssets[lockedAssetsCount].confirmada = true;
                    lockedAssetsCount++;
                } else {
                    Serial.printf(">>> VEREDITO VÁLIDO (ID %u): PERDEU ou QUARTO CHEIO para o ativo %s\n", idTransacaoRecebida, ativo_mac.c_str());
                }

                // Remove o alvo da lista de potenciais, pois a disputa acabou
                potentialTargets[i] = potentialTargets[potentialTargetsCount - 1];
                potentialTargetsCount--;
                
                return; // Encontrou e processou, pode sair
            }
        }
        Serial.printf(">>> VEREDITO IGNORADO! Nenhuma disputa ativa encontrada para o ID %u e MAC %s.\n", idTransacaoRecebida, ativo_mac.c_str());
        return;
    }

    // --- LÓGICA DE COMANDO DE RESET (Atualizada para multi-ativo) ---
    if (topicStr == MQTT_INDIVIDUAL_COMMAND_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        if (doc.containsKey("type") && strcmp(doc["type"], "command") == 0 && doc.containsKey("data")) {
            if (strcmp(doc["data"]["name"], "RESET_STATE") == 0) {
            
                Serial.println(">>> Comando 'RESET_STATE' recebido! Notificando saída de todos os ativos locais.");

                // 1. ANTES de limpar o estado, notifica a saída de cada ativo
                for (int i = 0; i < lockedAssetsCount; i++) {
                    Serial.printf("  - Enviando evento OUT para o ativo travado: %s\n", lockedAssets[i].mac.c_str());
                    // Envia o evento 'OUT' sem se preocupar com a média de RSSI ou WiFi. O servidor vai entender.
                    enviarEventoHTTP(lockedAssets[i].mac.c_str(), "OUT", -100, 0, 0); 
                }

                // 2. AGORA, limpa o estado local
                Serial.println(">>> Estado local limpo.");
                lockedAssetsCount = 0;
            }

            if (strcmp(doc["data"]["name"], "REBOOT") == 0) {
                Serial.println(">>> Comando 'REBOOT' recebido! Reiniciando em 3 segundos...");
                delay(3000);
                ESP.restart();
            }

            if (strcmp(doc["data"]["name"], "RSSI_TEST") == 0) {
                Serial.println(">>> Comando 'RSSI_TEST' recebido!");
                runRssiTest(); // Chama a função de teste
            }

            if (strcmp(doc["data"]["name"], "FETCH_CONFIG") == 0) {
                Serial.println(">>> Comando 'FETCH_CONFIG' individual recebido! Buscando novas configurações.");
                checkInitialState();
            }
        return;
        }
    }

    // --- LÓGICA DE COMANDO GLOBAL (Mantida do seu código) ---
    if (topicStr == MQTT_COMMAND_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        if (doc.containsKey("command") && strcmp(doc["command"], "fetch_config") == 0) {
            Serial.println(">>> Comando 'fetch_config' recebido! Buscando novas configurações.");
            checkInitialState();
        }
        return;
    }

    // --- LÓGICA DA LISTA DE ATIVOS DISPONÍVEIS (Mantida do seu código e adaptada) ---
    if (topicStr == MQTT_ASSET_LIST_TOPIC) {
        // --- LÓGICA DE ATUALIZAÇÃO SEGURA (SEM MEMCPY) ---
        Serial.println("Recebida nova lista de alvos. Sincronizando estado...");
        
        StaticJsonDocument<2048> doc;
        deserializeJson(doc, payload, length);
        JsonArray macArray = doc.as<JsonArray>();

        TargetAsset newPotentialTargets[MAX_AVAILABLE_ASSETS];
        int newCount = 0;

        for (JsonVariant v : macArray) {
            if (newCount >= MAX_AVAILABLE_ASSETS) break;
            String mac = v.as<String>();

            bool found_existing = false;
            for(int i=0; i < potentialTargetsCount; i++) {
                if (potentialTargets[i].mac.equalsIgnoreCase(mac) && potentialTargets[i].emDisputa) {
                    // Cópia segura, objeto por objeto
                    newPotentialTargets[newCount++] = potentialTargets[i];
                    found_existing = true;
                    Serial.printf("  - Estado de disputa para %s PRESERVADO.\n", mac.c_str());
                    break;
                }
            }

            if (!found_existing) {
                newPotentialTargets[newCount] = TargetAsset();
                newPotentialTargets[newCount].mac = mac;
                newCount++;
            }
        }

        // --- SUBSTITUIÇÃO DO MEMCPY POR UM FOR LOOP SEGURO ---
        potentialTargetsCount = newCount;
        for (int i = 0; i < potentialTargetsCount; i++) {
            potentialTargets[i] = newPotentialTargets[i];
        }
        // --- FIM DA SUBSTITUIÇÃO ---

        Serial.printf("Sincronização concluída. %d alvos na lista.\n", potentialTargetsCount);
        
        if (potentialTargetsCount == 0 && !inEmergencyCooldown) {
            Serial.println("\n>>> ALERTA: Lista de alvos vazia recebida! ...");
            inEmergencyCooldown = true;
            emergencyCooldownStart = millis();
        }
        return;
    }
}

// =================== SETUP E LOOP PRINCIPAL ===================
void setup() {
    Serial.begin(115200);
    randomSeed(analogRead(0));
    lerConfiguracoes();

    MQTT_VERDICT_TOPIC = "wyrd/rtls/esp/" + String(idESP) + "/verdict";
    MQTT_INDIVIDUAL_COMMAND_TOPIC = "wyrd/rtls/esp/" + String(idESP) + "/command";
    TRIGGER_HOTSPOT_SSID = "RTLS_" + String(idESP);
    MQTT_HEARTBEAT_TOPIC = "wyrd/rtls/esp/heartbeat/" + String(idESP);

    Serial.println("\n==============================");
    Serial.println("= EMBARCADO RTLS v0.3.4    =");
    Serial.println("==============================");

    WiFi.mode(WIFI_STA);
    WiFi.disconnect();
    delay(100);

    Serial.printf("Procurando pelo Hotspot de Configuração '%s'...\n", TRIGGER_HOTSPOT_SSID.c_str());
    bool triggerFound = false;
    unsigned long startTime = millis();
    while (millis() - startTime < TRIGGER_SCAN_TIMEOUT) {
        int n = WiFi.scanNetworks();
        for (int i = 0; i < n; ++i) {
            if (WiFi.SSID(i) == TRIGGER_HOTSPOT_SSID) {
                triggerFound = true;
                break;
            }
        }
        if (triggerFound) break;
        delay(500);
    }
    WiFi.scanDelete();

    if (triggerFound) {
        Serial.println("Hotspot encontrado! Iniciando Portal de Configuração...");
        WiFiManager wm;
        wm.setConfigPortalTimeout(0);
        wm.setSaveConfigCallback(saveConfigCallback);
        WiFiManagerParameter custom_server_ip("server_ip", "IP Servidor HTTP", SERVER_IP, 16);
        WiFiManagerParameter custom_server_port("server_port", "Porta Servidor HTTP", SERVER_PORT_STR, 6);
        WiFiManagerParameter custom_mqtt_host("mqtt_host", "IP Broker MQTT", MQTT_BROKER_HOST, 16);
        WiFiManagerParameter custom_mqtt_port("mqtt_port", "Porta Broker MQTT", MQTT_BROKER_PORT_STR, 6);
        wm.addParameter(&custom_server_ip);
        wm.addParameter(&custom_server_port);
        wm.addParameter(&custom_mqtt_host);
        wm.addParameter(&custom_mqtt_port);
        String portalAPName = "CONFIG_RTLS_" + String(idESP);
        if (wm.startConfigPortal(portalAPName.c_str())) {
            strcpy(SERVER_IP, custom_server_ip.getValue());
            strcpy(SERVER_PORT_STR, custom_server_port.getValue());
            strcpy(MQTT_BROKER_HOST, custom_mqtt_host.getValue());
            strcpy(MQTT_BROKER_PORT_STR, custom_mqtt_port.getValue());
            SERVER_PORT = atoi(SERVER_PORT_STR);
            MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
            salvarConfiguracoes();
            Serial.println("Configurações salvas. Reiniciando...");
            delay(5000);
            ESP.restart();
        } else {
            Serial.println("Portal fechado. Reiniciando...");
            delay(3000);
            ESP.restart();
        }
    } else {
        Serial.println("Iniciando em modo de operação normal.");
        WiFi.begin();
        unsigned long wifiStartTime = millis();
        while (WiFi.status() != WL_CONNECTED) {
            if(millis() - wifiStartTime > 20000) {
                Serial.println("Falha ao conectar ao Wi-Fi. Reiniciando...");
                delay(3000);
                ESP.restart();
            }
            delay(500);
        }
        Serial.println("Wi-Fi conectado!");

        syncTimeFromServer();
        BLEDevice::init("");
        BLEScan* pBLEScan = BLEDevice::getScan();
        pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
        pBLEScan->setActiveScan(true);
        mqttClient.setServer(MQTT_BROKER_HOST, MQTT_BROKER_PORT);
        mqttClient.setCallback(mqtt_callback);
        checkInitialState();
        if (mqttClient.connected()) {
            mqttClient.publish(MQTT_HEARTBEAT_TOPIC.c_str(), "");
            Serial.println(">>> Heartbeat inicial enviado.");
            isInitialized = true; // Se tudo correu bem, marcamos como inicializado
        } else {
            Serial.println("AVISO: Falha na conexão MQTT inicial. A ESP tentará novamente.");
            isInitialized = false; // Se falhou, marcamos para tentar de novo
        }
    }
}

void loop() {
    unsigned long agora = millis(); // Usar uma variável local 'agora'

    if (!isInitialized) {
        if (agora - lastInitAttemptMillis > INIT_RETRY_INTERVAL_MS) {
            Serial.println(">>> A inicialização ainda não foi concluída. Tentando sincronizar novamente...");
            
            // Tenta refazer os passos críticos da inicialização
            syncTimeFromServer();
            checkInitialState();
            
            // Tenta reconectar ao MQTT e subscrever aos tópicos
            if (!mqttClient.connected()) {
                reconnect_mqtt();
            }
            
            // Se agora estiver tudo OK, marca como inicializado
            if (mqttClient.connected()) {
                Serial.println(">>> Sincronização bem-sucedida! Operação normal retomada.");
                isInitialized = true;
            }
            
            lastInitAttemptMillis = agora; // Atualiza o temporizador da tentativa
        }
    }

    // --- LÓGICA DE HEARTBEAT ADICIONADA ---
    if (agora - lastHeartbeatMillis > HEARTBEAT_INTERVAL_MS) {
        if (mqttClient.connected()) {
            // Envia uma mensagem vazia. O tópico já diz quem somos e que estamos vivos.
            mqttClient.publish(MQTT_HEARTBEAT_TOPIC.c_str(), ""); 
            Serial.println(">>> Heartbeat enviado.");
        }
        lastHeartbeatMillis = agora;
    }

    if (inEmergencyCooldown && (millis() - emergencyCooldownStart > EMERGENCY_COOLDOWN_MS)) {
        Serial.println(">>> Cooldown de emergência finalizado.");
        inEmergencyCooldown = false;
    }

    if (!mqttClient.connected()) {
        reconnect_mqtt();
    }
    mqttClient.loop();
    BLEDevice::getScan()->start(BEACON_SCAN_TIME, false);
    processarLogicaAtivo();
    delay(500);
}


// Função auxiliar para remover um ativo da lista de travados
void removerAtivoTravado(int index) {
    if (index < 0 || index >= lockedAssetsCount) return;
    
    Serial.printf("Removendo ativo %s da lista de presentes.\n", lockedAssets[index].mac.c_str());
    for (int i = index; i < lockedAssetsCount - 1; i++) {
        lockedAssets[i] = lockedAssets[i + 1];
    }
    lockedAssetsCount--;
}

// =================== LÓGICA DE DECISÃO PRINCIPAL ===================
void processarLogicaAtivo() {
    unsigned long agora = millis();

    // 1. Lógica de SAÍDA (itera sobre a lista de presentes) - SEM MUDANÇAS
    for (int i = lockedAssetsCount - 1; i >= 0; i--) { // Usando o laço mais seguro
        LockedAsset &asset = lockedAssets[i];
        if (!asset.isFull) continue;

        int soma = 0;
        for (int k = 0; k < RSSI_HISTORY_SIZE; k++) {
            soma += asset.rssiHistory[k];
        }
        int mediaRSSI = soma / RSSI_HISTORY_SIZE;
        Serial.printf("Monitorando ativo presente: %s | Média RSSI: %d\n", asset.mac.c_str(), mediaRSSI);

        if (mediaRSSI < RSSI_THRESHOLD) {
            if (asset.inicioInerciaSaida == 0) {
                asset.inicioInerciaSaida = agora;
            }
        } else {
            asset.inicioInerciaSaida = 0;
        }

        if (asset.inicioInerciaSaida > 0 && (agora - asset.inicioInerciaSaida > INERCIA_SAIDA)) {
            Serial.printf(">>> Inércia de SAÍDA atingida para %s. Enviando evento OUT.\n", asset.mac.c_str());
            if (enviarEventoHTTP(asset.mac.c_str(), "OUT", mediaRSSI, WiFi.RSSI(), 0)) {
                removerAtivoTravado(i);
            } else {
                Serial.println("Falha ao reportar SAÍDA. Tentará novamente.");
            }
        }
    }

    // 2. Lógica de ENTRADA (agora processa TODOS os alvos elegíveis)
    for (int i = 0; i < potentialTargetsCount; i++) {
        TargetAsset &target = potentialTargets[i];

        if (target.emDisputa && (agora - target.inicioDisputa > VEREDICT_TIMEOUT_MS)) {
            Serial.printf("!!! TIMEOUT DE VEREDITO para %s. Cancelando disputa local.\n", target.mac.c_str());
            target.emDisputa = false;      // Libera o ativo para novas tentativas
            target.inicioInercia = 0;     // Reseta a inércia para ele poder ser reavaliado
            target.idTransacao = 0;
            target.inicioDisputa = 0;
        }

        if (target.emDisputa) continue; 
        if (!target.isFull) continue;

        int soma = 0;
        for (int j = 0; j < RSSI_HISTORY_SIZE; j++) soma += target.rssiHistory[j];
        target.mediaRSSI = soma / RSSI_HISTORY_SIZE;
        Serial.printf("  - Alvo disponível: %s | Média RSSI: %d\n", target.mac.c_str(), target.mediaRSSI);
        if (target.mediaRSSI > RSSI_THRESHOLD) {
            if (target.inicioInercia == 0) target.inicioInercia = agora;
        } else {
            target.inicioInercia = 0;
        }

        // Verifica se a inércia foi atingida
        if (target.inicioInercia > 0 && (agora - target.inicioInercia > INERCIA_CHEGADA)) {
            Serial.printf(">>> Inércia de CHEGADA atingida para %s. Enviando evento GET.\n", target.mac.c_str());

            uint32_t mac_hash = 0;
            for(int j=0; j < target.mac.length(); j++) {
                mac_hash = 31 * mac_hash + target.mac.charAt(j);
            }

            uint32_t esp_hash = 0;
            const char* espIdStr = idESP;
            for(int j=0; espIdStr[j] != '\0'; j++) {
                esp_hash = 31 * esp_hash + espIdStr[j];
            }

            uint32_t transacao = (mac_hash << 1) ^ esp_hash + millis();

            if (enviarEventoHTTP(target.mac.c_str(), "GET", target.mediaRSSI, WiFi.RSSI(), transacao)) {
                target.emDisputa = true;
                target.idTransacao = transacao; 
                target.inicioDisputa = millis(); 
                Serial.printf("Disputa iniciada para %s com ID OTIMIZADO %u.\n", target.mac.c_str(), transacao);
            }
        }
    }
    Serial.println("-----------------------");
}

// =================== FUNÇÕES AUXILIARES ===================
void runRssiTest() {
    Serial.println(">>> Iniciando teste de RSSI com dados em memória...");

    StaticJsonDocument<1024> doc;
    doc["esp_id"] = idESP; // Adiciona o ID da ESP ao objeto principal
    JsonArray report = doc.createNestedArray("report");

    Serial.printf("  - Reportando %d ativos presentes (locked)...\n", lockedAssetsCount);
    for (int i = 0; i < lockedAssetsCount; i++) {
        JsonObject obj = report.createNestedObject();
        obj["mac"] = lockedAssets[i].mac;
        
        int soma = 0;
        int count = lockedAssets[i].isFull ? RSSI_HISTORY_SIZE : lockedAssets[i].index;
        if (count > 0) {
            for (int k = 0; k < count; k++) {
                soma += lockedAssets[i].rssiHistory[k];
            }
            obj["rssi"] = soma / count;
        } else {
            obj["rssi"] = -100; // Valor padrão se não houver leituras
        }
        obj["status"] = "Presente";
    }

    // 2. Adiciona os alvos disponíveis que a ESP está a monitorizar
    Serial.printf("  - Reportando %d alvos disponíveis (potential)...\n", potentialTargetsCount);
    for (int i = 0; i < potentialTargetsCount; i++) {
        // Só reporta os que já tiveram pelo menos uma leitura de RSSI
        if (potentialTargets[i].index > 0 || potentialTargets[i].isFull) {
            JsonObject obj = report.createNestedObject();
            obj["mac"] = potentialTargets[i].mac;
            obj["rssi"] = potentialTargets[i].mediaRSSI; // Usa a média já calculada
            obj["status"] = "Disponível";
        }
    }
    
    String payload;
    serializeJson(doc, payload);

    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/rssi-report";
        
        http.begin(serverUrl);
        http.addHeader("Content-Type", "application/json");

        Serial.println(">>> Enviando relatório de teste de RSSI via HTTP POST...");
        int httpResponseCode = http.POST(payload);

        if (httpResponseCode > 0) {
            Serial.printf(">>> Resposta do servidor: %d\n", httpResponseCode);
        } else {
            Serial.printf("ERRO no envio do POST: %s\n", http.errorToString(httpResponseCode).c_str());
        }
        http.end();
    } else {
        Serial.println("ERRO: Não foi possível enviar o relatório. Wi-Fi desconectado.");
    }
}

void checkInitialState() {
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/esp/" + String(idESP) + "/config";
    Serial.printf("Buscando configuração inicial em: %s\n", serverUrl.c_str());

    if (!http.begin(serverUrl)) {
        Serial.println("Falha ao iniciar HTTP para config.");
        return;
    }

    int httpResponseCode = http.GET();
    if (httpResponseCode == 200) {
        String payload = http.getString();
        Serial.printf("Configuração recebida: %s\n", payload.c_str());
        StaticJsonDocument<1024> doc; // Aumentar o tamanho para a lista
        deserializeJson(doc, payload);

        // --- MUDANÇA AQUI: Processa a lista de MACs ---
        if (doc.containsKey("macs_beacons")) {
            JsonArray macArray = doc["macs_beacons"].as<JsonArray>();
            lockedAssetsCount = 0; // Limpa a lista atual
            for (JsonVariant v : macArray) {
                if (lockedAssetsCount < MAX_LOCKED_ASSETS) {
                    String mac = v.as<String>();
                    lockedAssets[lockedAssetsCount].mac = mac;
                    lockedAssets[lockedAssetsCount].confirmada = true;
                    // Reseta o histórico para começar a monitorar do zero
                    memset(lockedAssets[lockedAssetsCount].rssiHistory, 0, sizeof(lockedAssets[lockedAssetsCount].rssiHistory));
                    lockedAssets[lockedAssetsCount].index = 0;
                    lockedAssets[lockedAssetsCount].isFull = false;
                    lockedAssets[lockedAssetsCount].inicioInerciaSaida = 0;
                    lockedAssetsCount++;
                }
            }
            Serial.printf("%d ativos pré-carregados no estado.\n", lockedAssetsCount);
        }

        if (doc.containsKey("rssi_threshold")) RSSI_THRESHOLD = doc["rssi_threshold"];
        if (doc.containsKey("inercia_chegada")) INERCIA_CHEGADA = doc["inercia_chegada"];
        if (doc.containsKey("inercia_saida")) INERCIA_SAIDA = doc["inercia_saida"];
    } else {
        Serial.printf("Erro na busca de config: %d\n", httpResponseCode);
    }
    http.end();
}

bool enviarEventoHTTP(const char* mac, const char* status, int rssi, int wifi, uint32_t transacao_id) {
    if (WiFi.status() != WL_CONNECTED) return false;
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/event";
    if (!http.begin(serverUrl)) return false;
    http.addHeader("Content-Type", "application/json");

    StaticJsonDocument<256> doc;
    doc["esp_id"] = idESP;
    doc["ativo"] = mac;
    doc["status"] = status;
    doc["RSSI"] = rssi;
    doc["wifi"] = wifi;
    doc["data_on"] = obterDataAtualISO8601();

    if (strcmp(status, "GET") == 0) {
        doc["transacao_id"] = transacao_id;
    }

    String jsonPayload;
    serializeJson(doc, jsonPayload);
    int httpResponseCode = http.POST(jsonPayload);
    http.end();
    return (httpResponseCode == 202);
}

void reconnect_mqtt() {
    while (!mqttClient.connected()) {
        if (mqttClient.connect(idESP)) {
            mqttClient.subscribe(MQTT_ASSET_LIST_TOPIC);
            mqttClient.subscribe(MQTT_COMMAND_TOPIC);
            mqttClient.subscribe(MQTT_VERDICT_TOPIC.c_str());
            mqttClient.subscribe(MQTT_INDIVIDUAL_COMMAND_TOPIC.c_str());
        } else {
            delay(5000);
        }
    }
}

void syncTimeFromServer() {
    Serial.println("Tentando sincronizar o tempo com o servidor local...");
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/api/time";

    if (http.begin(serverUrl)) {
        int httpCode = http.GET();
        if (httpCode == 200) {
            String payload = http.getString();
            StaticJsonDocument<64> doc;
            deserializeJson(doc, payload);
            long unix_time = doc["unix_time"];

             if (unix_time > 0) {
                timeval tv = { unix_time, 0 };
                settimeofday(&tv, NULL);
                Serial.printf("Tempo sincronizado com sucesso via servidor local! Timestamp UTC: %ld\n", unix_time);

                struct tm timeinfo;
                setenv("TZ", "BRT+3", 1); // Define o fuso para Brasília (BRT, 3 horas a oeste de UTC)
                tzset();

                getLocalTime(&timeinfo);
                Serial.printf("Verificação da hora local (Brasília): %02d:%02d:%02d\n", timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec);
            }
        } else {
            Serial.printf("Falha ao buscar tempo do servidor. Código HTTP: %d\n", httpCode);
        }
        http.end();
    } else {
        Serial.println("Falha ao iniciar conexão HTTP para sincronização de tempo.");
    }
}

String obterDataAtualISO8601() {
    struct tm timeinfo;
    if (!getLocalTime(&timeinfo)) return "1970-01-01T00:00:00-03:00"; // Default com fuso
    
    char buffer[35]; // Aumentar um pouco o buffer para o fuso
    strftime(buffer, sizeof(buffer), "%Y-%m-%dT%H:%M:%S-03:00", &timeinfo);
    
    return String(buffer);
}
