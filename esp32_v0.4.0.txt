// ============ CONFIGURAÇÕES GLOBAIS ============
const char* idESP = "WRD00000002"; // <-- MUDE CONFORME A ESP
const char* FIRMWARE_VERSION = "0.4.0";

// --- Tópicos MQTT ---
const char* MQTT_COMMAND_TOPIC = "wyrd/rtls/esp/all/command";
String MQTT_SCAN_DATA_TOPIC = "";
String MQTT_INDIVIDUAL_COMMAND_TOPIC = "";
String MQTT_STATUS_TOPIC = ""; // Para o LWT

// --- Gatilho para Modo de Configuração ---
String TRIGGER_HOTSPOT_SSID = "";
const char* TRIGGER_HOTSPOT_PASSWORD = "rtlsconfig";
const unsigned long TRIGGER_SCAN_TIMEOUT = 5000;

// --- Constantes de Operação ---
#define BEACON_SCAN_DURATION_SEC 2.5
#define MAX_BEACONS_PER_SCAN 50
#define MAX_WHITELIST_SIZE 100
const unsigned long CONFIG_UPDATE_INTERVAL_MS = 300000; // 5 minutos
const unsigned long MQTT_REBOOT_TIMEOUT_MS = 600000;    // 10 minutos

// --- Configurações de Rede ---
char SERVER_IP[16]          = "10.0.0.124";
char SERVER_PORT_STR[6]     = "8080";
char MQTT_BROKER_HOST[16]   = "10.0.0.124";
char MQTT_BROKER_PORT_STR[6]= "1883";
uint16_t SERVER_PORT        = 8080;
uint16_t MQTT_BROKER_PORT   = 1883;

// ============ OBJETOS GLOBAIS E ESTRUTURAS DE DADOS ============
WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient);
Preferences preferences;

// --- Estruturas de Dados ---
struct BeaconData { String mac; int rssi; };
BeaconData foundBeacons[MAX_BEACONS_PER_SCAN];
int beacon_count = 0;
String asset_whitelist[MAX_WHITELIST_SIZE];
int whitelist_count = 0;

// --- Temporizadores ---
unsigned long mqttConnectionLostTimestamp = 0;
unsigned long lastConfigUpdateTime = 0;

// ============ PROTÓTIPOS DE FUNÇÕES ============
void lerConfiguracoes();
void salvarConfiguracoes();
void mqtt_callback(char* topic, byte* payload, unsigned int length);
void syncTimeFromServer();
void runRssiTest();
void reconnect_mqtt();
void publishScanData();

// =================== LÓGICA DE CONFIGURAÇÃO ===================
void lerConfiguracoes() {
    preferences.begin("rtls-config", false);
    String saved_server_ip = preferences.getString("server_ip", "10.0.0.124");
    String saved_server_port = preferences.getString("server_port", "8080");
    String saved_mqtt_host = preferences.getString("mqtt_host", "10.0.0.124");
    String saved_mqtt_port = preferences.getString("mqtt_port", "1883");
    strcpy(SERVER_IP, saved_server_ip.c_str());
    strcpy(SERVER_PORT_STR, saved_server_port.c_str());
    strcpy(MQTT_BROKER_HOST, saved_mqtt_host.c_str());
    strcpy(MQTT_BROKER_PORT_STR, saved_mqtt_port.c_str());
    SERVER_PORT = atoi(SERVER_PORT_STR);
    MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
    preferences.end();
}

void salvarConfiguracoes() {
    preferences.begin("rtls-config", false);
    preferences.putString("server_ip", SERVER_IP);
    preferences.putString("server_port", SERVER_PORT_STR);
    preferences.putString("mqtt_host", MQTT_BROKER_HOST);
    preferences.putString("mqtt_port", MQTT_BROKER_PORT_STR);
    preferences.end();
}

// =================== LÓGICA DE COMUNICAÇÃO ===================
class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) override {
        if (beacon_count >= MAX_BEACONS_PER_SCAN) return;

        String foundMac = advertisedDevice.getAddress().toString().c_str();
        
        for (int i = 0; i < whitelist_count; i++) {
            if (asset_whitelist[i].equalsIgnoreCase(foundMac)) {
                foundBeacons[beacon_count].mac = foundMac;
                foundBeacons[beacon_count].rssi = advertisedDevice.getRSSI();
                beacon_count++;
                return; // Encontrou, pode parar de procurar na lista
            }
        }
    }
};

void mqtt_callback(char* topic, byte* payload, unsigned int length) {
    LOGLN("------------------------------------");
    LOGF("Mensagem recebida no tópico: %s\n", topic);
    String topicStr = String(topic);

    if (topicStr == MQTT_INDIVIDUAL_COMMAND_TOPIC || topicStr == MQTT_COMMAND_TOPIC) {
        StaticJsonDocument<256> doc;
        deserializeJson(doc, payload, length);

        String command;
        if (doc.containsKey("command")) {
            command = doc["command"].as<String>();
        } else if (doc.containsKey("data") && doc["data"].containsKey("name")) {
            command = doc["data"]["name"].as<String>();
        }

        if (command == "RESET_STATE") {
            LOGLN(">>> Comando 'RESET_STATE' recebido! (Ação não mais necessária na v0.4.1).");
        } else if (command == "REBOOT") {
            LOGLN(">>> Comando 'REBOOT' recebido! Reiniciando em 3 segundos...");
            delay(3000);
            ESP.restart();
        } else if (command == "RSSI_TEST") {
            LOGLN(">>> Comando 'RSSI_TEST' recebido!");
            runRssiTest();
        } else if (command == "FETCH_CONFIG" || command == "fetch_config") {
            LOGLN(">>> Comando de reconfiguração recebido!");
            if (get_configuration_from_server()){
                lastConfigUpdateTime = millis();
            } 
        }
        return;
    }
}

void publishScanData() {
    if (beacon_count == 0 || !mqttClient.connected()) {
        return;
    }

    StaticJsonDocument<4096> doc;
    doc["timestamp"] = time(nullptr);
    JsonArray beaconsArray = doc.createNestedArray("beacons");
    for (int i = 0; i < beacon_count; i++) {
        JsonObject beaconObj = beaconsArray.createNestedObject();
        beaconObj["mac"] = foundBeacons[i].mac;
        beaconObj["rssi"] = foundBeacons[i].rssi;
    }

    String payload;
    serializeJson(doc, payload);

    if (mqttClient.publish(MQTT_SCAN_DATA_TOPIC.c_str(), payload.c_str(), false)) {
        LOGF("Publicando %d beacons via MQTT...\n", beacon_count);
    } else {
        LOGLN("Falha ao enfileirar mensagem MQTT para publicação.");
    }
}

// =================== SETUP E LOOP PRINCIPAL ===================
void setup() {
    Serial.begin(115200);
    
    lerConfiguracoes();

    MQTT_SCAN_DATA_TOPIC = "wyrd/rtls/esp/" + String(idESP) + "/scan_data";
    MQTT_INDIVIDUAL_COMMAND_TOPIC = "wyrd/rtls/esp/" + String(idESP) + "/command";
    TRIGGER_HOTSPOT_SSID = "RTLS_" + String(idESP);

    LOGLN("\n=================================");
    LOGLN("= EMBARCADO RTLS v0.4.1         =");
    LOGLN("= MODO: SENSOR COLABORATIVO     =");
    LOGLN("=================================");

    WiFi.mode(WIFI_STA);
    WiFi.disconnect(true);
    delay(100);

    LOGF("Procurando pelo Hotspot de Configuração '%s'...\n", TRIGGER_HOTSPOT_SSID.c_str());
    bool triggerFound = false;
    unsigned long startTime = millis();
    while (millis() - startTime < TRIGGER_SCAN_TIMEOUT) {
        int n = WiFi.scanNetworks();
        for (int i = 0; i < n; ++i) {
            if (WiFi.SSID(i) == TRIGGER_HOTSPOT_SSID) {
                triggerFound = true;
                break;
            }
        }
        if (triggerFound) break;
        delay(500);
    }
    WiFi.scanDelete();

    if (triggerFound) {
        LOGLN("Hotspot encontrado! Iniciando Portal de Configuração...");
        WiFiManager wm;
        wm.setConfigPortalTimeout(180);
        WiFiManagerParameter custom_server_ip("server_ip", "IP Servidor HTTP", SERVER_IP, 16);
        WiFiManagerParameter custom_server_port("server_port", "Porta Servidor HTTP", SERVER_PORT_STR, 6);
        WiFiManagerParameter custom_mqtt_host("mqtt_host", "IP Broker MQTT", MQTT_BROKER_HOST, 16);
        WiFiManagerParameter custom_mqtt_port("mqtt_port", "Porta Broker MQTT", MQTT_BROKER_PORT_STR, 6);
        wm.addParameter(&custom_server_ip);
        wm.addParameter(&custom_server_port);
        wm.addParameter(&custom_mqtt_host);
        wm.addParameter(&custom_mqtt_port);

        String portalAPName = "CONFIG_RTLS_" + String(idESP);
        if (wm.startConfigPortal(portalAPName.c_str())) {
            strcpy(SERVER_IP, custom_server_ip.getValue());
            strcpy(SERVER_PORT_STR, custom_server_port.getValue());
            strcpy(MQTT_BROKER_HOST, custom_mqtt_host.getValue());
            strcpy(MQTT_BROKER_PORT_STR, custom_mqtt_port.getValue());
            SERVER_PORT = atoi(SERVER_PORT_STR);
            MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
            salvarConfiguracoes();
            LOGLN("Configurações salvas. Reiniciando...");
            delay(3000);
            ESP.restart();
        } else {
            LOGLN("Portal fechado. Reiniciando...");
            delay(3000);
            ESP.restart();
        }
    } else {
        LOGLN("Iniciando em modo de operação normal.");
        WiFi.begin();
        
        unsigned long startWifiTime = millis();
        while (WiFi.status() != WL_CONNECTED) {
            if (millis() - startWifiTime > 20000) {
                LOGLN("Falha ao conectar Wi-Fi. Reiniciando...");
                delay(1000); ESP.restart();
            }
        }
    }
    
    LOGLN("Wi-Fi Conectado!");

    // O resto do setup só acontece se não entramos no modo portal.
    syncTimeFromServer();

    delay(500);

    unsigned long startHandshakeTime = millis();
    while (!get_configuration_from_server()) {
        LOGLN("Tentando handshake novamente em 5 segundos...");
        delay(5000);

        if (millis() - startHandshakeTime > 60000) { // Timeout de 1 minuto
            LOGLN("!!! Falha ao obter configuração do servidor por 1 minuto. Reiniciando. !!!");
            delay(1000);
            ESP.restart();
        }
    }

    lastConfigUpdateTime = millis(); // Marca a hora da última config bem-sucedida

    BLEDevice::init("");
    BLEScan* pBLEScan = BLEDevice::getScan();
    pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
    pBLEScan->setActiveScan(true);

    mqttClient.setServer(MQTT_BROKER_HOST, MQTT_BROKER_PORT);
    mqttClient.setKeepAlive(60);
    mqttClient.setCallback(mqtt_callback);
}

void loop() {

    if (WiFi.status() != WL_CONNECTED) {
        LOGLN("Conexão Wi-Fi perdida. Tentando reconectar...");
        WiFi.disconnect();
        WiFi.reconnect();
        unsigned long startAttemptTime = millis();
        while (WiFi.status() != WL_CONNECTED && millis() - startAttemptTime < 10000) {
            delay(500);
        }
        if (WiFi.status() != WL_CONNECTED) {
            LOGLN("Falha ao reconectar Wi-Fi. Reiniciando ESP...");
            delay(1000);
            ESP.restart();
        }
        LOGLN("Wi-Fi reconectado!");
    }
    if (!mqttClient.connected()) {
        if (mqttConnectionLostTimestamp == 0) {
            mqttConnectionLostTimestamp = millis();
            LOGLN("Conexão MQTT perdida. Iniciando temporizador de reboot...");
        } else {
            if (millis() - mqttConnectionLostTimestamp > MQTT_REBOOT_TIMEOUT_MS) {
                LOGLN("!!! Falha ao reconectar ao MQTT por 10 minutos. Forçando reboot de autocorreção. !!!");
                delay(1000);
                ESP.restart();
            }
        }
        reconnect_mqtt();
    } else {
        if (mqttConnectionLostTimestamp > 0) {
            LOGLN("Conexão MQTT reestabelecida!");
            mqttConnectionLostTimestamp = 0;
        }
    }
    
    if (mqttClient.connected()) {
        mqttClient.loop();
        if (millis() - lastConfigUpdateTime > CONFIG_UPDATE_INTERVAL_MS) {
            if (get_configuration_from_server()){
                lastConfigUpdateTime = millis();
            } 
        }
        beacon_count = 0;
        BLEDevice::getScan()->start(BEACON_SCAN_DURATION_SEC, false);
        publishScanData(); 
    }

    delay(500); // Mantém um ciclo de aproximadamente 2 segundos
}

// =================== FUNÇÕES AUXILIARES ===================
bool get_configuration_from_server() {
    LOGLN("Iniciando handshake com o servidor...");
    HTTPClient http;

    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/api/esp/handshake";
    serverUrl += "?id_esp=" + String(idESP);
    serverUrl += "&mac=" + WiFi.macAddress();
    serverUrl += "&ip=" + WiFi.localIP().toString();
    serverUrl += "&fw=" + String(FIRMWARE_VERSION);

    bool success = false;
    if (http.begin(serverUrl)) {
        int httpCode = http.GET();
        if (httpCode == 200) {
            DynamicJsonDocument doc(4096);
            deserializeJson(doc, http.getString());
            JsonArray whitelistArray = doc["whitelist"].as<JsonArray>();
            
            whitelist_count = 0; 
            for (JsonVariant v : whitelistArray) {
                if (whitelist_count < MAX_WHITELIST_SIZE) {
                    asset_whitelist[whitelist_count++] = v.as<String>();
                }
            }
            LOGF("Handshake bem-sucedido. Whitelist com %d ativos recebida.\n", whitelist_count);
            success = true; // SUCESSO!
        } else {
            LOGF("Falha no handshake. Código HTTP: %d\n", httpCode);
        }
        http.end();
    } else {
        LOGLN("Falha ao iniciar conexão HTTP para handshake.");
    }
    return success;
}


void runRssiTest() {
    LOGLN(">>> Iniciando teste de RSSI de varredura única (5 segundos)...");
    beacon_count = 0;
    BLEDevice::getScan()->start(5, false);

    LOGF("  - Scan de teste concluído. %d beacons encontrados.\n", beacon_count);

    StaticJsonDocument<4096> doc;
    doc["esp_id"] = idESP;
    JsonArray report = doc.createNestedArray("report");

    for (int i = 0; i < beacon_count; i++) {
        JsonObject obj = report.createNestedObject();
        obj["mac"] = foundBeacons[i].mac;
        obj["rssi"] = foundBeacons[i].rssi;
    }

    String payload;
    serializeJson(doc, payload);

    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/rssi-report";
        http.begin(serverUrl);
        http.addHeader("Content-Type", "application/json");
        int httpResponseCode = http.POST(payload);
        if (httpResponseCode > 0) {
            LOGF(">>> Relatório enviado. Resposta do servidor: %d\n", httpResponseCode);
        } else {
            LOGF("ERRO no envio do POST: %s\n", http.errorToString(httpResponseCode).c_str());
        }
        http.end();
    } else {
        LOGLN("ERRO: Não foi possível enviar o relatório. Wi-Fi desconectado.");
    }
}

void reconnect_mqtt() {
    int retries = 0;
    while (!mqttClient.connected() && retries < 5) {
        retries++;
        LOGF("Tentativa %d de conexão MQTT...\n", retries);
        
        String lwt_topic = "wyrd/rtls/esp/" + String(idESP) + "/status";
        const char* lwt_payload = "offline"; 
        bool will_retain = true; 
        uint8_t will_qos = 1; 
        
        LOGF("Configurando Last Will em: %s\n", lwt_topic.c_str());
        
        if (mqttClient.connect(idESP, "wyrd_user", "Wyrd2025", lwt_topic.c_str(), will_qos, will_retain, lwt_payload)) {
            LOGLN("Conectado ao Broker e LWT configurado!");
            
            mqttClient.publish(lwt_topic.c_str(), "online", true);

            mqttClient.subscribe(MQTT_COMMAND_TOPIC);
            mqttClient.subscribe(MQTT_INDIVIDUAL_COMMAND_TOPIC.c_str());
            LOGLN("Subscrito aos tópicos de comando.");
        } else {
            LOGF("Falhou, rc=%d. Tentando novamente em 3 segundos\n", mqttClient.state());
            delay(3000);
        }
    }
}

void syncTimeFromServer() {
    LOGLN("Tentando sincronizar o tempo com o servidor local...");
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/api/time";
    if (http.begin(serverUrl)) {
        int httpCode = http.GET();
        if (httpCode == 200) {
            String payload = http.getString();
            StaticJsonDocument<128> doc;
            deserializeJson(doc, payload);
            long unix_time = doc["unix_time"];
            if (unix_time > 0) {
                timeval tv = { unix_time, 0 };
                settimeofday(&tv, NULL);
                LOGF("Tempo sincronizado com sucesso! Timestamp UTC: %ld\n", unix_time);
            }
        } else {
            LOGF("Falha ao buscar tempo do servidor. Código HTTP: %d\n", httpCode);
        }
        http.end();
    } else {
        LOGLN("Falha ao iniciar conexão HTTP para sincronização de tempo.");
    }
}
