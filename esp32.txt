// ===================================================================================
// = BAXTER - FIRMWARE FINAL COM ARQUITETURA DE TÓPICOS UNIFICADA                      =
// ===================================================================================
//
// Este firmware é projetado para um dispositivo ESP32 que atua como um sensor de
// localização em um quarto. Ele escaneia por beacons Bluetooth (BLE) de baixa energia
// que estão atrelados a camas e se comunica com um servidor central para determinar
// a localização de cada cama em tempo real.
//
// PRINCIPAIS FUNCIONALIDADES:
// 1. Conectividade Wi-Fi com portal de configuração.
// 2. Comunicação via MQTT para receber comandos e listas de alvos.
// 3. Scanner Bluetooth para detectar a proximidade de beacons.
// 4. Envio de eventos (entrada/saída) para o servidor via HTTP POST.
// 5. Lógica de "veredito" para evitar que múltiplas ESPs compitam pela mesma cama.
// 6. Capacidade de ter seu estado resetado remotamente pelo servidor.
//
// ===================================================================================

#include <WiFi.h>
#include <WiFiManager.h>
#include <Preferences.h>
#include <BLEDevice.h>

#include <HTTPClient.h>

// --- Identificador Fixo ---
// Este ID deve ser único para cada placa ESP32.
const char* idESP = "WRD00000001"; // <-- MUDE CONFORME A ESP

// --- Tópicos MQTT ---
// Tópico público onde o servidor publica a lista de MACs de beacons de camas disponíveis.
const char* MQTT_BED_LIST_TOPIC = "wyrd/baxter/beds/available";
const char* MQTT_COMMAND_TOPIC = "wyrd/baxter/esp/all/command";
String MQTT_INDIVIDUAL_TOPIC = ""; 

// --- Parâmetros do GET de Emergência por Evento ---
const unsigned long EMERGENCY_COOLDOWN_MS = 60000; // 1 minuto (60.000 ms)
unsigned long emergencyCooldownStart = 0;
bool inEmergencyCooldown = false;

// --- GATILHO PARA MODO DE CONFIGURAÇÃO ---
String TRIGGER_HOTSPOT_SSID = ""; // Nome do Wi-Fi temporário para entrar em modo de configuração.
const char* TRIGGER_HOTSPOT_PASSWORD = "baxterconfig";
const unsigned long TRIGGER_SCAN_TIMEOUT = 5000; // Tempo em ms que a ESP procura pelo hotspot.

// --- Variáveis de configuração (carregadas da memória NVS ou portal) ---
char SERVER_IP[16]            = "10.0.0.149";
char SERVER_PORT_STR[6]       = "8000";
char MQTT_BROKER_HOST[16]     = "10.0.0.149";
char MQTT_BROKER_PORT_STR[6]  = "1883";
uint16_t SERVER_PORT          = 8000;
uint16_t MQTT_BROKER_PORT     = 1883;

// --- Parâmetros de Lógica de Detecção ---
int RSSI_THRESHOLD            = -60;   // Limite de força do sinal BLE para considerar uma "entrada".
unsigned long INERCIA_CHEGADA = 500;   // Tempo em ms que o sinal precisa ficar acima do threshold para confirmar uma entrada.
unsigned long INERCIA_SAIDA   = 15000; // Tempo em ms que o sinal precisa ficar abaixo do threshold para confirmar uma saída.
const unsigned long VEREDICT_TIMEOUT_MS = 30000; // 30 segundos
unsigned long inicioAguardandoVeredito = 0;
#define RSSI_HISTORY_SIZE       3      // Número de leituras de RSSI para calcular a média e suavizar o sinal.
#define BEACON_SCAN_TIME        1      // Duração em segundos de cada ciclo de scan BLE.


// --- Configurações de Tempo (NTP) ---
const char* NTP_SERVER          = "pool.ntp.org";
const long  GMT_OFFSET_SEC      = -3 * 3600; // Offset para o fuso horário de Brasília (UTC-3).
const int   DAYLIGHT_OFFSET_SEC = 0;

// ============ OBJETOS GLOBAIS E ESTRUTURAS DE DADOS ============
WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient);
Preferences preferences; // Objeto para salvar/ler configurações da memória flash (NVS).
bool shouldSaveConfig = false;
uint32_t idTransacaoAguardada = 0;

// Estrutura para armazenar informações de um beacon sendo rastreado (antes de ser confirmado).
struct TargetBed {
    String mac = "";
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInercia = 0;
    int mediaRSSI = -100;
};
// Estrutura para armazenar a cama que está "travada" nesta ESP.
struct LockedBed {
    String mac = "";
    bool confirmada = false;        // true se o servidor enviou 'WIN'.
    bool aguardandoVeredito = false; // true após enviar um 'GET', esperando a resposta do servidor.
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInerciaSaida = 0;
} lockedBed;

#define MAX_AVAILABLE_BEDS 50
TargetBed potentialTargets[MAX_AVAILABLE_BEDS];
int potentialTargetsCount = 0;

// ============ PROTÓTIPOS DE FUNÇÕES ============
void lerConfiguracoes();
void salvarConfiguracoes();
void saveConfigCallback();
void checkInitialState();
void mqtt_callback(char* topic, byte* payload, unsigned int length);
String obterDataAtualISO8601();

void processarLogicaCama();

void reconnect_mqtt();

// Lê as configurações de IP/Porta da memória NVS.
void lerConfiguracoes() {
    preferences.begin("baxter-config", false);
    String saved_server_ip = preferences.getString("server_ip", "192.168.0.5");
    String saved_server_port = preferences.getString("server_port", "8000");
    String saved_mqtt_host = preferences.getString("mqtt_host", "192.168.0.5");
    String saved_mqtt_port = preferences.getString("mqtt_port", "1883");
    
    strcpy(SERVER_IP, saved_server_ip.c_str());
    strcpy(SERVER_PORT_STR, saved_server_port.c_str());
    strcpy(MQTT_BROKER_HOST, saved_mqtt_host.c_str());
    strcpy(MQTT_BROKER_PORT_STR, saved_mqtt_port.c_str());

    SERVER_PORT = atoi(SERVER_PORT_STR);
    MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
    
    preferences.end();
    Serial.println("Configurações carregadas da memória NVS.");
}

// Salva as configurações na memória NVS.
void salvarConfiguracoes() {
    preferences.begin("baxter-config", false);
    preferences.putString("server_ip", SERVER_IP);
    preferences.putString("server_port", SERVER_PORT_STR);
    preferences.putString("mqtt_host", MQTT_BROKER_HOST);
    preferences.putString("mqtt_port", MQTT_BROKER_PORT_STR);
    preferences.end();
    Serial.println("Configurações salvas na memória NVS.");
}

// Callback do WiFiManager para sinalizar que novas configurações devem ser salvas.
void saveConfigCallback () {
    Serial.println("Callback de salvar configuração foi acionado.");
    shouldSaveConfig = true;
}

// Callback para cada dispositivo Bluetooth encontrado durante o scan.
class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) override {
        String foundMac = advertisedDevice.getAddress().toString().c_str();

        // Se a ESP já está travada em uma cama, só nos importamos com o RSSI dessa cama.
        if (lockedBed.confirmada) {
            if (foundMac.equalsIgnoreCase(lockedBed.mac)) {
                lockedBed.rssiHistory[lockedBed.index] = advertisedDevice.getRSSI();
                lockedBed.index = (lockedBed.index + 1) % RSSI_HISTORY_SIZE;
                if (lockedBed.index == 0) lockedBed.isFull = true;
            }
            return;
        }

        // Se não, adicionamos o RSSI a qualquer alvo potencial que corresponda ao MAC.
        for (int i = 0; i < potentialTargetsCount; i++) {
            if (foundMac.equalsIgnoreCase(potentialTargets[i].mac)) {
                TargetBed &target = potentialTargets[i];
                target.rssiHistory[target.index] = advertisedDevice.getRSSI();
                target.index = (target.index + 1) % RSSI_HISTORY_SIZE;
                if (target.index == 0) target.isFull = true;
                break;
            }
        }
    }
};

// Função principal que trata TODAS as mensagens MQTT recebidas.
void mqtt_callback(char* topic, byte* payload, unsigned int length) {
    Serial.println("------------------------------------");
    Serial.printf("Mensagem recebida no tópico: %s\n", topic);
    
    String topicStr = String(topic);
    StaticJsonDocument<256> doc;
    DeserializationError error = deserializeJson(doc, payload, length);

    if (error) {
        Serial.print(F("Falha ao decodificar JSON: "));
        Serial.println(error.c_str());
        return;
    }

    // --- LÓGICA 1: Processar mensagens no CANAL INDIVIDUAL UNIFICADO ---
    if (topicStr == MQTT_INDIVIDUAL_TOPIC) {
        // Primeiro, lemos o campo "type" para saber o que fazer com a mensagem.
        const char* type = doc["type"];
        if (type == NULL) return; // Mensagem inválida.

        // CASO 1.1: A mensagem é um VEREDITO (WIN/LOSE)
        if (strcmp(type, "verdict") == 0) {
            JsonObject data = doc["data"];
            uint32_t idTransacaoRecebida = data["transacao_id"];
            
            // VALIDAÇÃO CRÍTICA: A ESP só age se o ID recebido for o que ela está esperando.
            if (lockedBed.aguardandoVeredito && idTransacaoRecebida == idTransacaoAguardada) {
                const char* status = data["status"];
                String cama = data["cama"];

                if (cama.equalsIgnoreCase(lockedBed.mac)) {
                    if (strcmp(status, "WIN") == 0) {
                        Serial.printf(">>> VEREDITO VÁLIDO (ID: %u): VENCEU! Travando na cama %s\n", idTransacaoRecebida, cama.c_str());
                        lockedBed.confirmada = true;
                    } else { // "LOSE"
                        Serial.printf(">>> VEREDITO VÁLIDO (ID: %u): PERDEU! Liberando a cama %s\n", idTransacaoRecebida, cama.c_str());
                        lockedBed = {};
                    }
                    lockedBed.aguardandoVeredito = false;
                    idTransacaoAguardada = 0; // Limpa o ID esperado.
                }
            } else {
                Serial.printf(">>> VEREDITO IGNORADO! ID esperado: %u, ID recebido: %u.\n", idTransacaoAguardada, idTransacaoRecebida);
            }
        }
        // CASO 1.2: A mensagem é um COMANDO (RESET_STATE)
        else if (strcmp(type, "command") == 0) {
            JsonObject data = doc["data"];
            const char* command_name = data["name"];
            if (strcmp(command_name, "RESET_STATE") == 0) {
                Serial.println(">>> Comando 'RESET_STATE' recebido! Limpando estado.");
                lockedBed = {}; // Ação: reseta a struct para o estado inicial.
            }
        }
        return;
    }
    
    // --- LÓGICA 2: Processar mensagens no CANAL DE COMANDO GERAL ---
    if (topicStr == MQTT_COMMAND_TOPIC) {
        const char* command = doc["command"];
        if (command != NULL && strcmp(command, "fetch_config") == 0) {
            Serial.println(">>> Comando GERAL 'fetch_config' recebido!");
            checkInitialState();
        }
        return;
    }

    // --- LÓGICA 3: Processar a LISTA DE CAMAS DISPONÍVEIS ---
    if (topicStr == MQTT_BED_LIST_TOPIC) {
        // Se a ESP já está ocupada, ela ignora novas listas para evitar confusão.
        if (lockedBed.confirmada || lockedBed.aguardandoVeredito) {
            Serial.printf("ESP travada ou aguardando veredito. Ignorando nova lista.\n");
            return;
        }
        
        potentialTargetsCount = 0;
        JsonArray macArray = doc.as<JsonArray>();

        Serial.println("Inicializando nova lista de alvos potenciais:");
        for (JsonVariant v : macArray) {
            if (potentialTargetsCount < MAX_AVAILABLE_BEDS) {
                String mac = v.as<String>();
                potentialTargets[potentialTargetsCount] = TargetBed();
                potentialTargets[potentialTargetsCount].mac = mac;
                Serial.printf("  - Alvo %d: %s\n", potentialTargetsCount, mac.c_str());
                potentialTargetsCount++;
            }
        }
        Serial.println("------------------------------------");

        // --- LÓGICA DO GET DE EMERGÊNCIA POR EVENTO ---
        // Se a lista de alvos veio vazia E não estamos em cooldown...
        if (potentialTargetsCount == 0 && !inEmergencyCooldown) {
            Serial.println("\n>>> ALERTA: Lista de alvos vazia recebida! Possível dessincronização.");
            Serial.println(">>> Disparando GET de emergência para re-sincronizar...");
            
            // Pede o estado e as configs para o servidor. Se o servidor achar que
            // a ESP deveria estar com uma cama, esta chamada irá corrigir o estado.
            checkInitialState(); 

            // Inicia o cooldown para evitar spam de requisições.
            inEmergencyCooldown = true;
            emergencyCooldownStart = millis();
            Serial.println(">>> Cooldown de 1 minuto iniciado.");
        }
    }
}

// =================== SETUP E LOOP PRINCIPAL ======================
// =================================================================
void setup() {
    Serial.begin(115200);
    lerConfiguracoes();
    
    // Constrói dinamicamente os nomes dos tópicos específicos desta ESP.
    MQTT_INDIVIDUAL_TOPIC = "wyrd/baxter/esp/" + String(idESP) + "/command";
    TRIGGER_HOTSPOT_SSID = "BAXTER_" + String(idESP);

    Serial.println("\n\n==================");
    Serial.println("= EMBARCADO BAXTER INICIANDO =");
    Serial.println("==================");

    WiFi.mode(WIFI_STA);
    WiFi.disconnect();
    delay(100);

    // Lógica para detectar se deve entrar em modo de configuração.
    Serial.printf("Procurando pelo Hotspot de Configuração '%s' por %d segundos...\n", TRIGGER_HOTSPOT_SSID.c_str(), TRIGGER_SCAN_TIMEOUT / 1000);
    bool triggerFound = false;
    unsigned long startTime = millis();
    while (millis() - startTime < TRIGGER_SCAN_TIMEOUT) {
        int n = WiFi.scanNetworks();
        if (n > 0) {
            for (int i = 0; i < n; ++i) {
                if (WiFi.SSID(i) == TRIGGER_HOTSPOT_SSID) {
                    triggerFound = true;
                    break;
                }
            }
        }
        if (triggerFound) break;
        delay(500);
        Serial.print(".");
    }
    Serial.println();
    WiFi.scanDelete();

    // Se encontrou o hotspot, inicia o portal de configuração.
    if (triggerFound) {
        Serial.println("Iniciando Portal de Configuração...");
        WiFiManager wm;
        wm.setConfigPortalTimeout(0); 
        wm.setSaveConfigCallback(saveConfigCallback);

        WiFiManagerParameter custom_server_ip("server_ip", "IP Servidor HTTP", SERVER_IP, 16);
        WiFiManagerParameter custom_server_port("server_port", "Porta Servidor HTTP", SERVER_PORT_STR, 6);
        WiFiManagerParameter custom_mqtt_host("mqtt_host", "IP Broker MQTT", MQTT_BROKER_HOST, 16);
        WiFiManagerParameter custom_mqtt_port("mqtt_port", "Porta Broker MQTT", MQTT_BROKER_PORT_STR, 6);
        wm.addParameter(&custom_server_ip);
        wm.addParameter(&custom_server_port);
        wm.addParameter(&custom_mqtt_host);
        wm.addParameter(&custom_mqtt_port);

        String portalAPName = "CONFIG_BAXTER_" + String(idESP); 
        if (wm.startConfigPortal(portalAPName.c_str())) {
            strcpy(SERVER_IP, custom_server_ip.getValue());
            strcpy(SERVER_PORT_STR, custom_server_port.getValue());
            strcpy(MQTT_BROKER_HOST, custom_mqtt_host.getValue());
            strcpy(MQTT_BROKER_PORT_STR, custom_mqtt_port.getValue());
            SERVER_PORT = atoi(SERVER_PORT_STR);
            MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
            
            salvarConfiguracoes();
            Serial.println("Configurações salvas. Reiniciando em 5 segundos...");
            delay(5000);
            ESP.restart();
        } else {
            Serial.println("Portal fechado sem salvar. Reiniciando...");
            delay(3000);
            ESP.restart();
        }
    } 
    // Se não, inicia em modo de operação normal.
    else {
        Serial.println("\nHotspot de configuração não encontrado. Iniciando em modo de operação normal.");
        Serial.print("Conectando à rede Wi-Fi salva...");
        WiFi.begin();

        unsigned long wifiStartTime = millis();
        while (WiFi.status() != WL_CONNECTED) {
            if(millis() - wifiStartTime > 20000) {
                Serial.println("\nFalha ao conectar ao Wi-Fi. Reiniciando...");
                delay(3000);
                ESP.restart();
            }
            Serial.print(".");
            delay(500);
        }

        Serial.println("\nWi-Fi conectado com sucesso!");
        Serial.print("IP: ");
        Serial.println(WiFi.localIP());

        configurarNTP();
        
        BLEDevice::init("");
        BLEScan* pBLEScan = BLEDevice::getScan();
        pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
        pBLEScan->setActiveScan(true);

        mqttClient.setServer(MQTT_BROKER_HOST, MQTT_BROKER_PORT);
        mqttClient.setCallback(mqtt_callback);
        
        // Pergunta ao servidor qual era seu último estado conhecido.
        checkInitialState();
    }
}

// O loop principal da aplicação.
void loop() {
    if (inEmergencyCooldown && (millis() - emergencyCooldownStart > EMERGENCY_COOLDOWN_MS)) {
        Serial.println(">>> Cooldown de emergência finalizado. ESP pode pedir nova sincronização se necessário.");
        inEmergencyCooldown = false;
        emergencyCooldownStart = 0;
    }

    if (!mqttClient.connected()) {
        reconnect_mqtt();
    }
    mqttClient.loop(); // Processa mensagens MQTT recebidas.

    // Inicia um novo ciclo de scan BLE. O resultado será tratado no callback onResult.
    BLEDevice::getScan()->start(BEACON_SCAN_TIME, false);

    // Executa a lógica de decisão principal (verificar inércias, enviar eventos, etc.).
    processarLogicaCama();
    delay(500); // Pequena pausa para não sobrecarregar o processador.
}

// =================== LÓGICA DE DECISÃO PRINCIPAL =================
void processarLogicaCama() {
    unsigned long agora = millis();

    if (lockedBed.aguardandoVeredito && (agora - inicioAguardandoVeredito > VEREDICT_TIMEOUT_MS)) {
        Serial.println(">>> ERRO: Timeout! Veredito não recebido em 30 segundos. Resetando estado.");
        lockedBed = {}; // Reseta a trava
        idTransacaoAguardada = 0; // Limpa o ID da transação
        inicioAguardandoVeredito = 0;
        return;

    if (lockedBed.confirmada) {
        if (!lockedBed.isFull) { return; }

        int soma = 0;
        for (int i = 0; i < RSSI_HISTORY_SIZE; i++) { soma += lockedBed.rssiHistory[i]; }
        int mediaRSSI = soma / RSSI_HISTORY_SIZE;

        Serial.printf("Monitorando cama travada: %s | Média RSSI: %d\n", lockedBed.mac.c_str(), mediaRSSI);

        if (mediaRSSI < RSSI_THRESHOLD) {
            if (lockedBed.inicioInerciaSaida == 0) {
                Serial.printf("Sinal fraco detectado. Iniciando contagem de inércia de SAÍDA.\n");
                lockedBed.inicioInerciaSaida = agora;
            }
        } else {
            if (lockedBed.inicioInerciaSaida > 0) {
                Serial.println("Sinal forte restaurado. Cancelando contagem de inércia de SAÍDA.");
                lockedBed.inicioInerciaSaida = 0;
            }
        }

         if (lockedBed.inicioInerciaSaida > 0 && (agora - lockedBed.inicioInerciaSaida > INERCIA_SAIDA)) {
            Serial.printf(">>> Inércia de SAÍDA atingida. Tratando como SAÍDA.\n");
            bool sucesso = enviarEventoHTTP(lockedBed.mac.c_str(), "OUT", mediaRSSI, WiFi.RSSI());
            if (sucesso) {
                Serial.println("SAÍDA reportada com sucesso. Liberando trava.");
                lockedBed = {};
            } else {
                Serial.println("Falha ao reportar SAÍDA. Manterá a trava, mas resetando inércia para evitar flood.");
                lockedBed.inicioInerciaSaida = 0;
            }
        }
        return; 
    }

    for (int i = 0; i < potentialTargetsCount; i++) {
        TargetBed &target = potentialTargets[i];
        if (!target.isFull) continue;
        int soma = 0;
        for (int j = 0; j < RSSI_HISTORY_SIZE; j++) soma += target.rssiHistory[j];
        target.mediaRSSI = soma / RSSI_HISTORY_SIZE;
        Serial.printf("  - Alvo: %s | Média RSSI: %d\n", target.mac.c_str(), target.mediaRSSI);
        if (target.mediaRSSI > RSSI_THRESHOLD) {
            if (target.inicioInercia == 0) {
                target.inicioInercia = agora;
            }
        } else {
            target.inicioInercia = 0;
        }
    }
    Serial.println("------");
    
    int winnerIndex = -1;
    int bestRssiOfWinners = -1000;
    for (int i = 0; i < potentialTargetsCount; i++) {
        TargetBed &target = potentialTargets[i];
        if (target.inicioInercia > 0 && (agora - target.inicioInercia > INERCIA_CHEGADA)) {
            if (target.mediaRSSI > bestRssiOfWinners) {
                bestRssiOfWinners = target.mediaRSSI;
                winnerIndex = i;
            }
        }
    }

    if (winnerIndex != -1) {
        TargetBed &winner = potentialTargets[winnerIndex];
        bool sucesso = enviarEventoHTTP(winner.mac.c_str(), "GET", winner.mediaRSSI, WiFi.RSSI());
        
        if (sucesso) {
            lockedBed.mac = winner.mac;
            lockedBed.aguardandoVeredito = true;
            lockedBed.confirmada = false;
            lockedBed.index = 0;
            lockedBed.isFull = false;
            
            inicioAguardandoVeredito = millis();

            potentialTargetsCount = 0;
            Serial.printf("Evento de ENTRADA enviado. Aguardando veredito para a cama %s.\n", lockedBed.mac.c_str());
        } else {
            Serial.println("Falha ao reportar ENTRADA. A ESP continuará no modo de busca.");
        }
    }
}

void checkInitialState() {
    Serial.println("Verificando estado inicial e configurações com o servidor...");
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/esp/" + String(idESP) + "/config";
    
    if (!http.begin(serverUrl)) {
        Serial.println("Falha ao iniciar conexão. Usando configurações padrão.");
        return;
    }

    int httpResponseCode = http.GET();
    if (httpResponseCode == 200) {
        String payload = http.getString();
        Serial.println("Resposta do servidor: " + payload);
        StaticJsonDocument<256> doc;
        deserializeJson(doc, payload);

        const char* assignedMac = doc["mac_beacon"];
        if (assignedMac != NULL) {
            lockedBed.mac = String(assignedMac);
            lockedBed.confirmada = true;
            Serial.printf(">>> ESTADO RESTAURADO: ESP está travada na cama %s\n", lockedBed.mac.c_str());
        } else {
            Serial.println("Nenhuma cama atribuída a esta ESP. Iniciando no modo de busca.");
        }

        if (doc.containsKey("rssi_threshold")) {
            RSSI_THRESHOLD = doc["rssi_threshold"];
            Serial.printf(">>> Configuração remota: RSSI_THRESHOLD definido para %d\n", RSSI_THRESHOLD);
        }
        if (doc.containsKey("inercia_chegada")) {
            INERCIA_CHEGADA = doc["inercia_chegada"];
            Serial.printf(">>> Configuração remota: INERCIA_CHEGADA definida para %lu ms\n", INERCIA_CHEGADA);
        }
        if (doc.containsKey("inercia_saida")) {
            INERCIA_SAIDA = doc["inercia_saida"];
            Serial.printf(">>> Configuração remota: INERCIA_SAIDA definida para %lu ms\n", INERCIA_SAIDA);
        }

    } else {
        Serial.printf("Erro ao verificar estado. Código: %d. Usando configurações padrão.\n", httpResponseCode);
    }
    http.end();
}

bool enviarEventoHTTP(const char* mac, const char* status, int rssi, int wifi) {
    if (WiFi.status() != WL_CONNECTED) {
        Serial.println("Erro de Envio: Wi-Fi desconectado.");
        return false;
    }

    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/event";
    
    if (!http.begin(serverUrl)) {
        Serial.println("Erro de Envio: Falha ao iniciar conexão HTTP.");
        return false;
    }

    http.addHeader("Content-Type", "application/json");

    StaticJsonDocument<256> doc;
    doc["esp_id"] = idESP;
    doc["cama"] = mac; 
    doc["status"] = status;
    doc["RSSI"] = rssi;
    doc["wifi"] = wifi;
    doc["data_on"] = obterDataAtualISO8601();

    // ADIÇÃO: Se o evento for um GET, gera e adiciona um ID de transação.
    if (strcmp(status, "GET") == 0) {
        idTransacaoAguardada = random(1, 4294967295); // Gera um número aleatório de 32 bits.
        doc["transacao_id"] = idTransacaoAguardada;
        Serial.printf("Gerado novo ID de Transação para GET: %u\n", idTransacaoAguardada);
    }

    String jsonPayload;
    serializeJson(doc, jsonPayload);

    Serial.println("Enviando payload: " + jsonPayload);
    int httpResponseCode = http.POST(jsonPayload);

    // Servidores mais complexos podem responder 202 (Accepted) para eventos que
    // vão para uma fila de processamento, como a lógica de veredito.
    if (httpResponseCode == 202) { 
        Serial.printf("Sucesso! Evento aceito (202). Aguardando veredito.\n");
        http.end();
        return true;
    } else if (httpResponseCode > 0) {
        Serial.printf("Erro do Servidor. Código: %d\n", httpResponseCode);
        String response = http.getString();
        Serial.println(response);
        http.end();
        return false;
    } else {
        Serial.printf("Erro de conexão: %s\n", http.errorToString(httpResponseCode).c_str());
        http.end();
        return false;
    }
}

void reconnect_mqtt() {
    while (!mqttClient.connected()) {
        Serial.print("Tentando conectar ao Broker MQTT...");
        if (mqttClient.connect(idESP)) {
            Serial.println(" conectado!");
            mqttClient.subscribe(MQTT_BED_LIST_TOPIC);
            mqttClient.subscribe(MQTT_COMMAND_TOPIC); 
            mqttClient.subscribe(MQTT_INDIVIDUAL_TOPIC.c_str());
            Serial.println("Reconectado ao MQTT. Sincronizando estado com o servidor...");
            checkInitialState();
        } else {
            Serial.print(" falhou, rc=");
            Serial.print(mqttClient.state());
            Serial.println(" tentando novamente em 5 segundos");
            delay(5000);
        }
    }
}

String obterDataAtualISO8601() {
    struct tm timeinfo;
    if (!getLocalTime(&timeinfo)) {
        return "1970-01-01T00:00:00.000Z";
    }
    char buffer[30];
    strftime(buffer, sizeof(buffer), "%Y-%m-%dT%H:%M:%S.000Z", &timeinfo);
    return String(buffer);
}
