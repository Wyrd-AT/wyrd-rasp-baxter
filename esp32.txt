// ============ CONFIGURAÇÕES GLOBAIS ============
const char* idESP = "WRD00000001"; // <-- MUDE CONFORME A ESP

// --- Tópicos MQTT ---
const char* MQTT_BADGE_LIST_TOPIC = "wyrd/eritel/badges/available";
const char* MQTT_COMMAND_TOPIC = "wyrd/eritel/esp/all/command";
String MQTT_VERDICT_TOPIC = "";
String MQTT_INDIVIDUAL_COMMAND_TOPIC = "";

// --- Gatilho para Modo de Configuração ---
String TRIGGER_HOTSPOT_SSID = "";
const char* TRIGGER_HOTSPOT_PASSWORD = "eritelconfig";
const unsigned long TRIGGER_SCAN_TIMEOUT = 5000;

// --- Variáveis de configuração ---
char SERVER_IP[16]          = "10.0.0.149";
char SERVER_PORT_STR[6]     = "8000";
char MQTT_BROKER_HOST[16]   = "10.0.0.149";
char MQTT_BROKER_PORT_STR[6]= "1883";
uint16_t SERVER_PORT        = 8000;
uint16_t MQTT_BROKER_PORT   = 1883;

// --- Configurações de Lógica ---
int RSSI_THRESHOLD          = -60;
unsigned long INERCIA_CHEGADA = 500;
unsigned long INERCIA_SAIDA   = 15000;
#define RSSI_HISTORY_SIZE     3
#define BEACON_SCAN_TIME      1

// --- Constantes de Resiliência ---
const unsigned long VEREDICT_TIMEOUT_MS = 30000;
const unsigned long EMERGENCY_COOLDOWN_MS = 60000;

// --- Configurações NTP ---
const char* NTP_SERVER        = "pool.ntp.org";
const long  GMT_OFFSET_SEC    = -3 * 3600;
const int   DAYLIGHT_OFFSET_SEC = 0;

// ============ OBJETOS GLOBAIS E ESTRUTURAS DE DADOS ============
WiFiClient wifiClient;
PubSubClient mqttClient(wifiClient);
Preferences preferences;
bool shouldSaveConfig = false;

uint32_t idTransacaoAguardada = 0;
unsigned long inicioAguardandoVeredito = 0;
unsigned long emergencyCooldownStart = 0;
bool inEmergencyCooldown = false;
bool aguardandoVereditoGlobal = false; // <-- NOVO: Controla o estado de disputa

#define MAX_LOCKED_BADGES 10

struct TargetBadge {
    String mac = "";
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInercia = 0;
    int mediaRSSI = -100;
};

struct LockedBadge {
    String mac = "";
    bool confirmada = false;
    int rssiHistory[RSSI_HISTORY_SIZE];
    uint8_t index = 0;
    bool isFull = false;
    unsigned long inicioInerciaSaida = 0;
} lockedBadge;

LockedBadge lockedBadges[MAX_LOCKED_BADGES]; // <-- NOVO: Array para os crachás presentes
int lockedBadgesCount = 0;    

#define MAX_AVAILABLE_BADGES 50
TargetBadge potentialTargets[MAX_AVAILABLE_BADGES];
int potentialTargetsCount = 0;

// =================== LÓGICA DE CONFIGURAÇÃO ===================
void lerConfiguracoes() {
    preferences.begin("eritel-config", false);
    String saved_server_ip = preferences.getString("server_ip", "192.168.0.5");
    String saved_server_port = preferences.getString("server_port", "8000");
    String saved_mqtt_host = preferences.getString("mqtt_host", "192.168.0.5");
    String saved_mqtt_port = preferences.getString("mqtt_port", "1883");
    strcpy(SERVER_IP, saved_server_ip.c_str());
    strcpy(SERVER_PORT_STR, saved_server_port.c_str());
    strcpy(MQTT_BROKER_HOST, saved_mqtt_host.c_str());
    strcpy(MQTT_BROKER_PORT_STR, saved_mqtt_port.c_str());
    SERVER_PORT = atoi(SERVER_PORT_STR);
    MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
    preferences.end();
}

void salvarConfiguracoes() {
    preferences.begin("eritel-config", false);
    preferences.putString("server_ip", SERVER_IP);
    preferences.putString("server_port", SERVER_PORT_STR);
    preferences.putString("mqtt_host", MQTT_BROKER_HOST);
    preferences.putString("mqtt_port", MQTT_BROKER_PORT_STR);
    preferences.end();
}

void saveConfigCallback () {
    shouldSaveConfig = true;
}

// =================== LÓGICA DE COMUNICAÇÃO ===================
class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks {
    void onResult(BLEAdvertisedDevice advertisedDevice) override {
        String foundMac = advertisedDevice.getAddress().toString().c_str();

        // --- MUDANÇA AQUI: Verifica a lista de crachás travados ---
        for (int i = 0; i < lockedBadgesCount; i++) {
            if (foundMac.equalsIgnoreCase(lockedBadges[i].mac)) {
                lockedBadges[i].rssiHistory[lockedBadges[i].index] = advertisedDevice.getRSSI();
                lockedBadges[i].index = (lockedBadges[i].index + 1) % RSSI_HISTORY_SIZE;
                if (lockedBadges[i].index == 0) lockedBadges[i].isFull = true;
                return; // Encontrou o crachá, pode parar a busca
            }
        }

        // Se não for um crachá travado, verifica os alvos potenciais (lógica inalterada)
        if (aguardandoVereditoGlobal) return; // Não busca novos alvos durante uma disputa

        for (int i = 0; i < potentialTargetsCount; i++) {
            if (foundMac.equalsIgnoreCase(potentialTargets[i].mac)) {
                TargetBadge &target = potentialTargets[i];
                target.rssiHistory[target.index] = advertisedDevice.getRSSI();
                target.index = (target.index + 1) % RSSI_HISTORY_SIZE;
                if (target.index == 0) target.isFull = true;
                break;
            }
        }
    }
};

void mqtt_callback(char* topic, byte* payload, unsigned int length) {
    Serial.println("------------------------------------");
    Serial.printf("Mensagem recebida no tópico: %s\n", topic);
    String topicStr = String(topic);

    // --- LÓGICA DE VEREDITO (Atualizada para multi-crachá) ---
    if (topicStr == MQTT_VERDICT_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        uint32_t idTransacaoRecebida = doc["transacao_id"];

        // Usamos a flag global 'aguardandoVereditoGlobal'
        if (aguardandoVereditoGlobal && idTransacaoRecebida == idTransacaoAguardada) {
            const char* status = doc["status"];
            String cracha_mac = doc["cracha"];
            
            if (strcmp(status, "WIN") == 0 && lockedBadgesCount < MAX_LOCKED_BADGES) {
                Serial.printf(">>> VEREDITO VÁLIDO (ID %u): VENCEU! Travando no crachá %s\n", idTransacaoRecebida, cracha_mac.c_str());
                // Adiciona o novo crachá à lista de presentes
                lockedBadges[lockedBadgesCount] = {}; // Reseta a struct para garantir que está limpa
                lockedBadges[lockedBadgesCount].mac = cracha_mac;
                lockedBadges[lockedBadgesCount].confirmada = true;
                lockedBadgesCount++;
            } else {
                Serial.printf(">>> VEREDITO VÁLIDO (ID %u): PERDEU ou QUARTO CHEIO! Liberando o crachá %s\n", idTransacaoRecebida, cracha_mac.c_str());
            }
            // Reseta o estado de disputa
            aguardandoVereditoGlobal = false;
            idTransacaoAguardada = 0;
        } else {
            Serial.printf(">>> VEREDITO IGNORADO! ID esperado: %u, ID recebido: %u.\n", idTransacaoAguardada, idTransacaoRecebida);
        }
        return;
    }

    // --- LÓGICA DE COMANDO DE RESET (Atualizada para multi-crachá) ---
    if (topicStr == MQTT_INDIVIDUAL_COMMAND_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        if (doc.containsKey("type") && strcmp(doc["type"], "command") == 0 && doc.containsKey("data") && strcmp(doc["data"]["name"], "RESET_STATE") == 0) {
            Serial.println(">>> Comando 'RESET_STATE' recebido! Limpando estado.");
            // Limpa a lista de crachás e o estado de disputa
            lockedBadgesCount = 0;
            aguardandoVereditoGlobal = false;
        }
        return;
    }

    // --- LÓGICA DE COMANDO GLOBAL (Mantida do seu código) ---
    if (topicStr == MQTT_COMMAND_TOPIC) {
        StaticJsonDocument<128> doc;
        deserializeJson(doc, payload, length);
        if (doc.containsKey("command") && strcmp(doc["command"], "fetch_config") == 0) {
            Serial.println(">>> Comando 'fetch_config' recebido! Buscando novas configurações.");
            checkInitialState();
        }
        return;
    }

    // --- LÓGICA DA LISTA DE CRACHÁS DISPONÍVEIS (Mantida do seu código e adaptada) ---
    if (topicStr == MQTT_BADGE_LIST_TOPIC) {
        // Apenas adaptamos a condição de bloqueio para a nova flag global
        if (aguardandoVereditoGlobal) {
            return;
        }
        
        potentialTargetsCount = 0;
        StaticJsonDocument<2048> doc;
        deserializeJson(doc, payload, length);
        JsonArray macArray = doc.as<JsonArray>();

        // A sua lógica de emergência é ótima e foi mantida 100%
        if (macArray.size() == 0 && !inEmergencyCooldown) {
            Serial.println("\n>>> ALERTA: Lista de alvos vazia recebida! Possível dessincronização.");
            Serial.println(">>> Disparando GET de emergência para re-sincronizar...");
            checkInitialState();
            inEmergencyCooldown = true;
            emergencyCooldownStart = millis();
            Serial.println(">>> Cooldown de emergência de 1 minuto iniciado.");
        } else {
            for (JsonVariant v : macArray) {
                if (potentialTargetsCount < MAX_AVAILABLE_BADGES) {
                    String mac = v.as<String>();
                    potentialTargets[potentialTargetsCount] = TargetBadge();
                    potentialTargets[potentialTargetsCount].mac = mac;
                    // Serial.printf("  - Alvo %d: %s\n", potentialTargetsCount, mac.c_str()); // Descomente se precisar debugar
                    potentialTargetsCount++;
                }
            }
            Serial.printf("Nova lista com %d alvos recebida.\n", potentialTargetsCount);
        }
    }
}

// =================== SETUP E LOOP PRINCIPAL ===================
void setup() {
    Serial.begin(115200);
    lerConfiguracoes();

    MQTT_VERDICT_TOPIC = "wyrd/eritel/esp/" + String(idESP) + "/verdict";
    MQTT_INDIVIDUAL_COMMAND_TOPIC = "wyrd/eritel/esp/" + String(idESP) + "/command";
    TRIGGER_HOTSPOT_SSID = "ERITEL_" + String(idESP);

    Serial.println("\n==============================");
    Serial.println("= EMBARCADO ERITEL v0.2.3    =");
    Serial.println("==============================");

    WiFi.mode(WIFI_STA);
    WiFi.disconnect();
    delay(100);

    Serial.printf("Procurando pelo Hotspot de Configuração '%s'...\n", TRIGGER_HOTSPOT_SSID.c_str());
    bool triggerFound = false;
    unsigned long startTime = millis();
    while (millis() - startTime < TRIGGER_SCAN_TIMEOUT) {
        int n = WiFi.scanNetworks();
        for (int i = 0; i < n; ++i) {
            if (WiFi.SSID(i) == TRIGGER_HOTSPOT_SSID) {
                triggerFound = true;
                break;
            }
        }
        if (triggerFound) break;
        delay(500);
    }
    WiFi.scanDelete();

    if (triggerFound) {
        Serial.println("Hotspot encontrado! Iniciando Portal de Configuração...");
        WiFiManager wm;
        wm.setConfigPortalTimeout(0);
        wm.setSaveConfigCallback(saveConfigCallback);
        WiFiManagerParameter custom_server_ip("server_ip", "IP Servidor HTTP", SERVER_IP, 16);
        WiFiManagerParameter custom_server_port("server_port", "Porta Servidor HTTP", SERVER_PORT_STR, 6);
        WiFiManagerParameter custom_mqtt_host("mqtt_host", "IP Broker MQTT", MQTT_BROKER_HOST, 16);
        WiFiManagerParameter custom_mqtt_port("mqtt_port", "Porta Broker MQTT", MQTT_BROKER_PORT_STR, 6);
        wm.addParameter(&custom_server_ip);
        wm.addParameter(&custom_server_port);
        wm.addParameter(&custom_mqtt_host);
        wm.addParameter(&custom_mqtt_port);
        String portalAPName = "CONFIG_ERITEL_" + String(idESP);
        if (wm.startConfigPortal(portalAPName.c_str())) {
            strcpy(SERVER_IP, custom_server_ip.getValue());
            strcpy(SERVER_PORT_STR, custom_server_port.getValue());
            strcpy(MQTT_BROKER_HOST, custom_mqtt_host.getValue());
            strcpy(MQTT_BROKER_PORT_STR, custom_mqtt_port.getValue());
            SERVER_PORT = atoi(SERVER_PORT_STR);
            MQTT_BROKER_PORT = atoi(MQTT_BROKER_PORT_STR);
            salvarConfiguracoes();
            Serial.println("Configurações salvas. Reiniciando...");
            delay(5000);
            ESP.restart();
        } else {
            Serial.println("Portal fechado. Reiniciando...");
            delay(3000);
            ESP.restart();
        }
    } else {
        Serial.println("Iniciando em modo de operação normal.");
        WiFi.begin();
        unsigned long wifiStartTime = millis();
        while (WiFi.status() != WL_CONNECTED) {
            if(millis() - wifiStartTime > 20000) {
                Serial.println("Falha ao conectar ao Wi-Fi. Reiniciando...");
                delay(3000);
                ESP.restart();
            }
            delay(500);
        }
        Serial.println("Wi-Fi conectado!");

        configurarNTP();
        BLEDevice::init("");
        BLEScan* pBLEScan = BLEDevice::getScan();
        pBLEScan->setAdvertisedDeviceCallbacks(new MyAdvertisedDeviceCallbacks());
        pBLEScan->setActiveScan(true);
        mqttClient.setServer(MQTT_BROKER_HOST, MQTT_BROKER_PORT);
        mqttClient.setCallback(mqtt_callback);
        checkInitialState();
    }
}

void loop() {
    if (inEmergencyCooldown && (millis() - emergencyCooldownStart > EMERGENCY_COOLDOWN_MS)) {
        Serial.println(">>> Cooldown de emergência finalizado.");
        inEmergencyCooldown = false;
    }

    if (!mqttClient.connected()) {
        reconnect_mqtt();
    }
    mqttClient.loop();
    BLEDevice::getScan()->start(BEACON_SCAN_TIME, false);
    processarLogicaCracha();
    delay(500);
}


// Função auxiliar para remover um crachá da lista de travados
void removerCrachaTravado(int index) {
    if (index < 0 || index >= lockedBadgesCount) return;
    
    Serial.printf("Removendo crachá %s da lista de presentes.\n", lockedBadges[index].mac.c_str());
    for (int i = index; i < lockedBadgesCount - 1; i++) {
        lockedBadges[i] = lockedBadges[i + 1];
    }
    lockedBadgesCount--;
}

// =================== LÓGICA DE DECISÃO PRINCIPAL ===================
void processarLogicaCracha() {
    unsigned long agora = millis();

    // 1. Lógica de Timeout do Veredito (agora global)
    if (aguardandoVereditoGlobal && (agora - inicioAguardandoVeredito > VEREDICT_TIMEOUT_MS)) {
        Serial.println(">>> ERRO: Timeout! Veredito não recebido em 30 segundos. Resetando estado de disputa.");
        aguardandoVereditoGlobal = false;
        idTransacaoAguardada = 0;
        return;
    }
    if (aguardandoVereditoGlobal) {
        return; // Se está esperando um veredito, não faz mais nada
    }
    
    // 2. Lógica de SAÍDA (itera sobre a lista de presentes)
    for (int i = 0; i < lockedBadgesCount; i++) {
        LockedBadge &badge = lockedBadges[i];
        if (!badge.isFull) continue; // Só processa se tiver histórico de RSSI

        int soma = 0;
        for (int k = 0; k < RSSI_HISTORY_SIZE; k++) {
            soma += badge.rssiHistory[k];
        }
        int mediaRSSI = soma / RSSI_HISTORY_SIZE;
        Serial.printf("Monitorando crachá presente: %s | Média RSSI: %d\n", badge.mac.c_str(), mediaRSSI);

        if (mediaRSSI < RSSI_THRESHOLD) {
            if (badge.inicioInerciaSaida == 0) {
                badge.inicioInerciaSaida = agora;
            }
        } else {
            badge.inicioInerciaSaida = 0; // Sinal forte, reseta a inércia
        }

        if (badge.inicioInerciaSaida > 0 && (agora - badge.inicioInerciaSaida > INERCIA_SAIDA)) {
            Serial.printf(">>> Inércia de SAÍDA atingida para %s. Enviando evento OUT.\n", badge.mac.c_str());
            if (enviarEventoHTTP(badge.mac.c_str(), "OUT", mediaRSSI, WiFi.RSSI())) {
                removerCrachaTravado(i);
                i--; // Decrementa o índice para compensar a remoção
            } else {
                Serial.println("Falha ao reportar SAÍDA. Tentará novamente.");
            }
        }
    }

    // 3. Lógica de ENTRADA (busca por novos crachás)
    for (int i = 0; i < potentialTargetsCount; i++) {
        TargetBadge &target = potentialTargets[i];
        if (!target.isFull) continue;
        int soma = 0;
        for (int j = 0; j < RSSI_HISTORY_SIZE; j++) soma += target.rssiHistory[j];
        target.mediaRSSI = soma / RSSI_HISTORY_SIZE;
        Serial.printf("  - Alvo: %s | Média RSSI: %d\n", target.mac.c_str(), target.mediaRSSI);
        if (target.mediaRSSI > RSSI_THRESHOLD) {
            if (target.inicioInercia == 0) target.inicioInercia = agora;
        } else {
            target.inicioInercia = 0;
        }
    }
    Serial.println("------");

    int winnerIndex = -1;
    int bestRssiOfWinners = -1000;
    for (int i = 0; i < potentialTargetsCount; i++) {
        TargetBadge &target = potentialTargets[i];
        if (target.inicioInercia > 0 && (agora - target.inicioInercia > INERCIA_CHEGADA)) {
            if (target.mediaRSSI > bestRssiOfWinners) {
                bestRssiOfWinners = target.mediaRSSI;
                winnerIndex = i;
            }
        }
    }

    if (winnerIndex != -1) {
        String winnerMac = potentialTargets[winnerIndex].mac;

        // --- VERIFICAÇÃO ADICIONAL CRÍTICA ---
        bool jaPresente = false;
        for (int i = 0; i < lockedBadgesCount; i++) {
            if (winnerMac.equalsIgnoreCase(lockedBadges[i].mac)) {
                jaPresente = true;
                break;
            }
        }

        if (!jaPresente) {
            Serial.printf(">>> Vencedor detectado: %s. Enviando evento GET.\n", winnerMac.c_str());
            if (enviarEventoHTTP(winnerMac.c_str(), "GET", potentialTargets[winnerIndex].mediaRSSI, WiFi.RSSI())) {
                aguardandoVereditoGlobal = true;
                inicioAguardandoVeredito = millis();
                potentialTargetsCount = 0; // Limpa os alvos após iniciar a disputa
            }
        } else {
            Serial.printf("Vencedor (%s) já está na lista de presentes. Ignorando.\n", winnerMac.c_str());
        }
    }
}

// =================== FUNÇÕES AUXILIARES ===================
void checkInitialState() {
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/esp/" + String(idESP) + "/config";
    Serial.printf("Buscando configuração inicial em: %s\n", serverUrl.c_str());

    if (!http.begin(serverUrl)) {
        Serial.println("Falha ao iniciar HTTP para config.");
        return;
    }

    int httpResponseCode = http.GET();
    if (httpResponseCode == 200) {
        String payload = http.getString();
        Serial.printf("Configuração recebida: %s\n", payload.c_str());
        StaticJsonDocument<1024> doc; // Aumentar o tamanho para a lista
        deserializeJson(doc, payload);

        // --- MUDANÇA AQUI: Processa a lista de MACs ---
        if (doc.containsKey("macs_beacons")) {
            JsonArray macArray = doc["macs_beacons"].as<JsonArray>();
            lockedBadgesCount = 0; // Limpa a lista atual
            for (JsonVariant v : macArray) {
                if (lockedBadgesCount < MAX_LOCKED_BADGES) {
                    String mac = v.as<String>();
                    lockedBadges[lockedBadgesCount].mac = mac;
                    lockedBadges[lockedBadgesCount].confirmada = true;
                    // Reseta o histórico para começar a monitorar do zero
                    memset(lockedBadges[lockedBadgesCount].rssiHistory, 0, sizeof(lockedBadges[lockedBadgesCount].rssiHistory));
                    lockedBadges[lockedBadgesCount].index = 0;
                    lockedBadges[lockedBadgesCount].isFull = false;
                    lockedBadges[lockedBadgesCount].inicioInerciaSaida = 0;
                    lockedBadgesCount++;
                }
            }
            Serial.printf("%d crachás pré-carregados no estado.\n", lockedBadgesCount);
        }

        if (doc.containsKey("rssi_threshold")) RSSI_THRESHOLD = doc["rssi_threshold"];
        if (doc.containsKey("inercia_chegada")) INERCIA_CHEGADA = doc["inercia_chegada"];
        if (doc.containsKey("inercia_saida")) INERCIA_SAIDA = doc["inercia_saida"];
    } else {
        Serial.printf("Erro na busca de config: %d\n", httpResponseCode);
    }
    http.end();
}

bool enviarEventoHTTP(const char* mac, const char* status, int rssi, int wifi) {
    if (WiFi.status() != WL_CONNECTED) return false;
    HTTPClient http;
    String serverUrl = "http://" + String(SERVER_IP) + ":" + String(SERVER_PORT_STR) + "/event";
    if (!http.begin(serverUrl)) return false;
    http.addHeader("Content-Type", "application/json");
    StaticJsonDocument<256> doc;
    doc["esp_id"] = idESP;
    doc["cracha"] = mac;
    doc["status"] = status;
    doc["RSSI"] = rssi;
    doc["wifi"] = wifi;
    doc["data_on"] = obterDataAtualISO8601();

    if (strcmp(status, "GET") == 0) {
        idTransacaoAguardada = random(1, 4294967295);
        doc["transacao_id"] = idTransacaoAguardada;
        Serial.printf("Gerado novo ID de Transação para GET: %u\n", idTransacaoAguardada);
    }

    String jsonPayload;
    serializeJson(doc, jsonPayload);
    int httpResponseCode = http.POST(jsonPayload);
    http.end();
    return (httpResponseCode == 202);
}

void reconnect_mqtt() {
    while (!mqttClient.connected()) {
        if (mqttClient.connect(idESP)) {
            mqttClient.subscribe(MQTT_BADGE_LIST_TOPIC);
            mqttClient.subscribe(MQTT_COMMAND_TOPIC);
            mqttClient.subscribe(MQTT_VERDICT_TOPIC.c_str());
            mqttClient.subscribe(MQTT_INDIVIDUAL_COMMAND_TOPIC.c_str());
        } else {
            delay(5000);
        }
    }
}

void configurarNTP() {
    configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC, NTP_SERVER);
}

String obterDataAtualISO8601() {
    struct tm timeinfo;
    if (!getLocalTime(&timeinfo)) return "1970-01-01T00:00:00.000Z";
    char buffer[30];
    strftime(buffer, sizeof(buffer), "%Y-%m-%dT%H:%M:%S.000Z", &timeinfo);
    return String(buffer);
}
