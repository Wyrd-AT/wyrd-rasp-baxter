document.addEventListener('DOMContentLoaded', function () {
    // Lógica para controlar a abertura e fechamento do menu de ações
    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation(); // Impede que o clique feche o menu imediatamente
            
            // Fecha qualquer outro menu que esteja aberto
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });

            // Alterna a exibição do menu clicado
            this.nextElementSibling.classList.toggle('show');
        });
    });

    // Fecha o menu se o usuário clicar em qualquer outro lugar da tela
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });
});
</script>

        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }

                if (event.data === "ATUALIZAR_ESTADO") {
                    console.log("Recebido sinal para atualizar. Recarregando a página...");
                    // Adiciona um pequeno delay para garantir que o banco de dados foi atualizado
                    setTimeout(() => {
                        window.location.reload();
                    }, 500); // 500ms de delay
                }   

                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });

<script>
document.addEventListener('DOMContentLoaded', function () {

    // ==========================================================
    // SEÇÃO 0: FUNÇÃO UTILITÁRIA GLOBAL
    // ==========================================================
    async function populateSelect(url, selectElement, valueField, textField, prompt, selectedValue = null) {
        if (!selectElement) return;
        selectElement.innerHTML = `<option value="">Carregando...</option>`;
        try {
            const response = await fetch(url);
            if (!response.ok) throw new Error('A resposta da rede não foi OK');
            const data = await response.json();
            selectElement.innerHTML = `<option value="">${prompt}</option>`;
            data.forEach(item => {
                const option = document.createElement('option');
                option.value = item[valueField];
                option.textContent = item[textField];
                if (selectedValue && item[valueField] == selectedValue) {
                    option.selected = true;
                }
                selectElement.appendChild(option);
            });
        } catch (error) {
            console.error('Erro ao popular o select:', error);
            selectElement.innerHTML = `<option value="">Erro ao carregar</option>`;
        }
    }

    // ==========================================================
    // SEÇÃO 1: LÓGICA DO FORMULÁRIO DE CADASTRO/EDIÇÃO DE INSTÂNCIA
    // ==========================================================
    const instanceForm = document.getElementById('instance-form');
    if (instanceForm) {
        const isEditMode = !!("{{ equipamento_para_editar.id if equipamento_para_editar else '' }}");
        const tipoSelect = document.getElementById('inst_equipamento_tipo_id');
        const localizacaoSelect = document.getElementById('inst_localizacao_id');
        const bastidorSelect = document.getElementById('inst_bastidor_id');

    async function checkRfidStatus(tagId) {
        if (!tagId) { rfidStatusText.textContent = ""; saveInstanceBtn.disabled = true; return; }
        try {
            rfidStatusText.textContent = "Verificando status...";
            let apiUrl = `/api/rfid-status/${tagId}`;
            if (editModeEquipId) { apiUrl += `?current_equip_id=${editModeEquipId}`; }
            const response = await fetch(apiUrl);
            const statusResult = await response.json();
            rfidStatusText.textContent = statusResult.message;
            if (statusResult.status === 'ASSOCIADO') {
                rfidStatusText.style.color = '#333'; rfidStatusText.innerHTML += ` <strong>${statusResult.equipamento_nome}</strong>`;
                saveInstanceBtn.disabled = true;
            } else {
                rfidStatusText.style.color = '#198754'; saveInstanceBtn.disabled = false;
            }
        } catch (error) {
            rfidStatusText.textContent = "Não foi possível verificar o status."; rfidStatusText.style.color = '#333';
            saveInstanceBtn.disabled = true;
        }
    }

    function connectRfidSocket() {
        if (rfidSocket && rfidSocket.readyState === WebSocket.OPEN) return;
        const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        rfidSocket = new WebSocket(`${ws_protocol}//${window.location.host}/ws/rfid`);
        rfidSocket.onopen = () => { readTagBtn.disabled = false; checkReaderBtn.disabled = false; rfidStatusText.textContent = "Leitor pronto."; };
        rfidSocket.onmessage = (event) => {
            const data = JSON.parse(event.data);
            if (data.type === 'reader_status') { rfidStatusText.textContent = data.message; rfidStatusText.style.color = data.status === 'OK' ? '#198754' : '#333'; }
            if (data.type === 'scan_status' || data.type === 'scan_error') { rfidStatusText.textContent = data.message; if (data.message !== "Aguardando leitura da tag...") { readTagBtn.disabled = false; readTagBtn.textContent = "Ler Tag"; } }
            if (data.type === 'tag_scanned') { rfidInput.value = data.tag_id; checkRfidStatus(data.tag_id); readTagBtn.disabled = false; readTagBtn.textContent = "Ler Tag"; }
        };
        rfidSocket.onclose = () => { readTagBtn.disabled = true; checkReaderBtn.disabled = true; rfidStatusText.textContent = "Leitor desconectado."; };
    }

    if (readTagBtn) {
        readTagBtn.addEventListener('click', () => {
            if (!rfidSocket || rfidSocket.readyState !== WebSocket.OPEN) { connectRfidSocket(); return; }
            readTagBtn.disabled = true; readTagBtn.textContent = 'Lendo...'; rfidStatusText.textContent = "Aproxime a tag...";
            rfidSocket.send(JSON.stringify({ action: "read_single_tag" }));
        });
        checkReaderBtn.addEventListener('click', () => {
            if (rfidSocket && rfidSocket.readyState === WebSocket.OPEN) { rfidSocket.send(JSON.stringify({ action: "check_health" })); }
        });
        rfidInput.addEventListener('blur', () => { checkRfidStatus(rfidInput.value.trim()); });
        if (editModeEquipId) { checkRfidStatus(rfidInput.value.trim()); }
        connectRfidSocket();
    }

    async function createSimpleCrudManager(config) {
        const { entityName, apiUrl, tableBodyId, formId, renderRow, onUpdate } = config;
        const tableBody = document.getElementById(tableBodyId);
        const form = document.getElementById(formId);
        const editIdInput = document.getElementById('edit-tipo-id');
        const cancelBtn = document.getElementById('cancel-edit-tipo-btn');
        const loadItems = async () => {
            tableBody.innerHTML = `<tr><td colspan="2">Carregando...</td></tr>`;
            const response = await fetch(apiUrl); const items = await response.json(); tableBody.innerHTML = '';
            if (items.length === 0) { tableBody.innerHTML = `<tr><td colspan="2">Nenhum item.</td></tr>`; }
            else { items.forEach(item => tableBody.appendChild(renderRow(item))); }
        };
        form.addEventListener('submit', async (e) => {
            e.preventDefault();
            const data = Object.fromEntries(new FormData(form).entries());
            const response = await fetch(apiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
            if (response.ok) { form.reset(); await loadItems(); if(onUpdate) onUpdate(); }
            else { alert(`Erro: ${(await response.json()).detail}`); }
        });
        tableBody.addEventListener('click', async (e) => {
            const target = e.target.closest('button'); if (!target) return;
            const id = target.dataset.id;
            if (target.classList.contains('btn-delete')) {
                if (confirm(`Excluir ${entityName}?`)) {
                    const response = await fetch(`${apiUrl}/${id}`, { method: 'DELETE' });
                    if (response.ok) { await loadItems(); if(onUpdate) onUpdate(); }
                    else { alert(`Erro: ${(await response.json()).detail}`); }
                }
            }
            if (target.classList.contains('btn-edit')) {
                const currentName = target.dataset.name;
                const newName = prompt(`Digite o novo nome para "${currentName}":`, currentName);
                if (newName && newName.trim() && newName !== currentName) {
                    try {
                        const response = await fetch(`${apiUrl}/${id}`, {
                            method: 'PUT',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify({ nome: newName.trim() })
                        });
                        if (!response.ok) throw new Error((await response.json()).detail);
                        await loadItems();
                        if (onUpdate) onUpdate();
                    } catch (error) { alert(`Erro ao editar: ${error.message}`); }
                }
            }
        });
        await loadItems();
    }

        const loadItems = async () => {
            tableBody.innerHTML = `<tr><td colspan="2">Carregando...</td></tr>`;

                tableBody.innerHTML = '';
                if (items.length === 0) {
                    tableBody.innerHTML = `<tr><td colspan="2">Nenhum fabricante cadastrado.</td></tr>`;

            const method = editId ? 'PUT' : 'POST';
            const url = editId ? `${apiUrl}/${editId}` : apiUrl;

                resetFormToCreateMode();
                await loadItems();
                if (onUpdate) onUpdate(); // Avisa outras partes do sistema que a lista mudou

            }
        });

        tableBody.addEventListener('click', (e) => {
            const target = e.target.closest('button');
            if (!target) return;
            const id = target.dataset.id;

            // Lógica para o botão Excluir
            if (target.classList.contains('btn-delete')) {
                if (confirm('Tem certeza que deseja excluir este fabricante?')) {
                    fetch(`${apiUrl}/${id}`, { method: 'DELETE' })
                        .then(async response => {
                            if (!response.ok) throw new Error((await response.json()).detail);
                            loadItems();
                            if (onUpdate) onUpdate();
                        })
                        .catch(error => alert(`Erro ao excluir: ${error.message}`));
                }
            }

            if (target.classList.contains('btn-edit')) {
                const currentName = target.dataset.name;

                form.querySelector('[name="nome"]').value = currentName;

                cancelBtn.hidden = false;
            }
        });

        // 6. Lógica do botão Cancelar
        cancelBtn.addEventListener('click', resetFormToCreateMode);

        // 7. Inicialização
        await loadItems();
    }

        const updateLocalizacaoDropdown = async () => {
            await populateSelect('/api/localizacoes', localizacaoSelect, 'id', 'nome', 'Selecione...');
            const response = await fetch('/api/localizacoes');
            localizacoes = await response.json();
        };

        // 4. Função para carregar e renderizar os bastidores na tabela
        const loadItems = async () => {
            tableBody.innerHTML = `<tr><td colspan="3">Carregando...</td></tr>`;
            const response = await fetch('/api/bastidores');
            const items = await response.json();
            tableBody.innerHTML = '';
            items.forEach(item => {
                const loc = localizacoes.find(l => l.id === item.localizacao_id);
                const tr = document.createElement('tr');
                // O botão de editar agora armazena o objeto inteiro
                tr.innerHTML = `
                    <td>${item.codigo_bast}</td>
                    <td>${loc ? loc.nome : 'N/A'}</td>
                    <td class="actions-cell">
                        <button class="btn-secondary btn-sm btn-edit" data-item='${JSON.stringify(item)}'>Editar</button>
                        <button class="btn-danger btn-sm btn-delete" data-id="${item.id}">Excluir</button>
                    </td>`;
                tableBody.appendChild(tr);
            });
        };
        
        // 5. Lógica de submissão do formulário (CRIAR ou ATUALIZAR)
        form.addEventListener('submit', async (e) => {
            e.preventDefault();
            const data = {
                codigo_bast: form.querySelector('input[name="codigo_bast"]').value,
                localizacao_id: parseInt(localizacaoSelect.value)
            };
            const editId = editIdInput.value;

            const method = editId ? 'PUT' : 'POST';
            const url = editId ? `/api/bastidores/${editId}` : '/api/bastidores';
            
            try {
                const response = await fetch(url, { method: method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) });
                if (!response.ok) throw new Error((await response.json()).detail);
                resetFormToCreateMode();
                await loadItems();
            } catch (error) {
                alert(`Erro ao salvar: ${error.message}`);
            }
        });

        // 6. Lógica de clique na tabela (EDITAR ou EXCLUIR)
        tableBody.addEventListener('click', (e) => {
            const target = e.target.closest('button');
            if (!target) return;
            const id = target.dataset.id;

            // Lógica para o botão Excluir
            if (target.classList.contains('btn-delete')) {
                if (confirm('Excluir este bastidor?')) {
                    fetch(`/api/bastidores/${id}`, { method: 'DELETE' })
                        .then(async response => {
                            if (!response.ok) throw new Error((await response.json()).detail);
                            loadItems();
                        })
                        .catch(error => alert(`Erro ao excluir: ${error.message}`));
                }
            }

            // Lógica para o botão Editar
            if (target.classList.contains('btn-edit')) {
                const item = JSON.parse(target.dataset.item);

                // Preenche o formulário com os dados do item
                form.querySelector('[name="codigo_bast"]').value = item.codigo_bast;
                form.querySelector('[name="localizacao_id"]').value = item.localizacao_id;

                // Entra em modo de edição
                editIdInput.value = item.id;
                formTitle.textContent = `Editando Bastidor: ${item.codigo_bast}`;
                submitBtn.textContent = 'Atualizar';
                cancelBtn.hidden = false;
            }
        });

        // 7. Lógica do botão Cancelar
        cancelBtn.addEventListener('click', resetFormToCreateMode);

        // 8. Inicialização
        await updateLocalizacaoDropdown();
        await loadItems();

        // 9. Retorna a função de atualização para que outros possam chamá-la
        return { updateLocalizacaoDropdown };
    }
    
    // --- INICIALIZAÇÃO DE TODOS OS GERENCIADORES ---
    (async () => {
        const tipoManager = await manageEquipamentoTipos();
        const bastidorManager = await manageBastidores();
        
        await manageFabricantes({
            onUpdate: async () => { await tipoManager.updateFabricanteDropdown(); }
        });
        await manageLocalizacoes({
            onUpdate: async () => {
                populateSelect('/api/localizacoes', document.getElementById('inst_localizacao_id'), 'id', 'nome', 'Selecione');
                await bastidorManager.updateLocalizacaoDropdown();
            }
        });
    })();
});

                try {

// Variável global para guardar nosso ID de cliente do WebSocket
let myClientId = null;

document.addEventListener('DOMContentLoaded', function () {
    // Lógica para controlar a abertura e fechamento do modal
    const modal = document.getElementById('rssi-modal');
    const closeBtn = document.getElementById('modal-close-btn');
    if (modal && closeBtn) {
        closeBtn.onclick = () => { modal.style.display = "none"; };
        window.onclick = (event) => {
            if (event.target == modal) {
                modal.style.display = "none";
            }
        };
    }

            event.stopPropagation();
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });
            this.nextElementSibling.classList.toggle('show');
        });
    });
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));

                modalTitle.textContent = `Relatório de RSSI da ESP: ${esp_id} (Limite: ${threshold} dBm)`;
                
                let reportHtml = `<table class="data-table"><thead><tr><th>Nome do Ativo</th><th>MAC Address</th><th>RSSI</th></tr></thead><tbody>`;

                if (data.report && data.report.length > 0) {
                    data.report.sort((a, b) => b.rssi - a.rssi); // Ordena do mais forte para o mais fraco
                    data.report.forEach(beacon => {
                        const assetName = assetNameMap[beacon.mac.toLowerCase()] || '<i>Desconhecido</i>';
                        const rssiValue = beacon.rssi;
                        const rssiClass = rssiValue >= threshold ? 'rssi-strong' : 'rssi-weak';
                        
                        reportHtml += `<tr><td>${assetName}</td><td>${beacon.mac}</td><td class="${rssiClass}">${rssiValue} dBm</td></tr>`;
                    });
                } else {
                    reportHtml += '<tr><td colspan="3" style="text-align:center;">Nenhum beacon conhecido encontrado.</td></tr>';
                }

                reportHtml += '</tbody></table>';
                modalBody.innerHTML = reportHtml;
                modal.style.display = "block";
            }
        } catch (e) { /* Ignora outras mensagens que não sejam JSON */ }
    });
    // --- FIM DA NOVA LÓGICA ---
});

document.addEventListener('DOMContentLoaded', function () {

    // --- PARTE 1: LÓGICA DOS FILTROS DEPENDENTES ---
    const localizacaoFilter = document.getElementById('filter_localizacao');
    const bastidorFilter = document.getElementById('filter_bastidor');
    const filtersForm = document.getElementById('filters-form');

    if (localizacaoFilter && filtersForm) {
        localizacaoFilter.addEventListener('change', async function() {
            // Limpa o filtro de bastidor antes de submeter o formulário
            if (bastidorFilter) {
                bastidorFilter.value = "";
            }
            // Submete o formulário para recarregar a página com a localização filtrada
            filtersForm.submit();
        });
    }

    // --- PARTE 2: LÓGICA DO MENU DE AÇÕES E EXCLUSÃO ---
    const tableBody = document.querySelector('.data-table tbody');
    if (tableBody) {
        // Lógica para abrir/fechar o menu de três pontinhos
        tableBody.addEventListener('click', function(event) {
            const trigger = event.target.closest('.actions-menu-trigger');
            if (trigger) {
                event.stopPropagation();
                const currentDropdown = trigger.nextElementSibling;
                // Fecha outros menus que possam estar abertos
                document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                    if (d !== currentDropdown) d.classList.remove('show');
                });
                // Alterna o menu atual
                currentDropdown.classList.toggle('show');
            }
        });

        // Lógica para o botão DELETAR
        tableBody.addEventListener('click', async function(event) {
            const deleteButton = event.target.closest('.btn-delete');
            if (deleteButton) {
                const equipId = deleteButton.dataset.id;
                const equipName = deleteButton.dataset.name;
                if (confirm(`Tem certeza que deseja excluir o equipamento "${equipName}"? Esta ação não pode ser desfeita.`)) {
                    try {
                        const response = await fetch(`/api/equipamentos/instancia/${equipId}`, {
                            method: 'DELETE',
                        });
                        if (response.status === 204) {
                            alert('Equipamento excluído com sucesso.');
                            window.location.reload();
                        } else {
                            const errorData = await response.json();
                            throw new Error(errorData.detail || 'Erro ao excluir o equipamento.');
                        }
                    } catch (error) {
                        alert(`Erro: ${error.message}`);
                    }
                }
            }
        });
    }

    // Fecha todos os menus de ação se o usuário clicar em qualquer outro lugar
    window.addEventListener('click', () => {
        document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
    });
});
</script>

<script>
document.addEventListener('DOMContentLoaded', function () {
    // Script da Pistola RFID
    const startBtn = document.getElementById('start-scan-btn');
    const stopBtn = document.getElementById('stop-scan-btn');
    const cancelBtn = document.getElementById('cancel-scan-btn'); 
    const statusDiv = document.getElementById('scan-status');
    const tagsListDiv = document.getElementById('scanned-tags-list');
    const dcSelect = document.getElementById('datacenter-select');
    const countSpan = document.getElementById('scanned-tags-count');
    const checkBtn = document.getElementById('check-reader-status-btn');
    let scannedTagsCount = 0;

    function resetScanUI() {
        statusDiv.textContent = "Status: Conectado e pronto.";
        statusDiv.style.color = '#6c757d';
        statusDiv.style.fontWeight = 'normal';
        tagsListDiv.innerHTML = '<div class="muted">Aguardando início da leitura...</div>';
        scannedTagsCount = 0;
        countSpan.textContent = `(0)`;
        startBtn.disabled = false;
        stopBtn.disabled = true;
        cancelBtn.disabled = true;
    }

    if (startBtn) { 
        const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        const rfidSocket = new WebSocket(`${ws_protocol}//${window.location.host}/ws/rfid`);
        
        rfidSocket.onopen = () => { resetScanUI(); checkBtn.disabled = false; };
        rfidSocket.onclose = () => { 
            statusDiv.textContent = "Status: Desconectado. Recarregue a página.";
            startBtn.disabled = true; stopBtn.disabled = true; cancelBtn.disabled = true; checkBtn.disabled = true;
        };
        rfidSocket.onmessage = (event) => {
            const data = JSON.parse(event.data);
            if (data.type === 'reader_status') {
                statusDiv.textContent = data.message;
                statusDiv.style.color = data.status === 'OK' ? '#198754' : '#333';
                statusDiv.style.fontWeight = data.status === 'ERROR' ? 'bold' : 'normal';
            }
            if (data.type === 'scan_status' || data.type === 'scan_error') {
                statusDiv.textContent = `Status: ${data.message}`;
                if (data.message.toLowerCase().includes("finalizada") || data.message.toLowerCase().includes("erro") || data.message.toLowerCase().includes("cancelada")) {
                    resetScanUI();
                }
            }
            if (data.type === 'tag_scanned') {
                if (tagsListDiv.querySelector('.muted')) { tagsListDiv.innerHTML = ''; }
                const tagElement = document.createElement('code');
                tagElement.textContent = data.tag_id;
                tagsListDiv.appendChild(tagElement);
                scannedTagsCount++;
                countSpan.textContent = `(${scannedTagsCount})`;
            }
        };

        checkBtn.addEventListener('click', () => {
            if (rfidSocket.readyState === WebSocket.OPEN) {
                rfidSocket.send(JSON.stringify({ action: "check_health" }));
            }
        });
        startBtn.addEventListener('click', () => {
            const datacenterId = dcSelect.value;
            if (!datacenterId) { alert("Por favor, selecione um datacenter primeiro."); return; }
            if (rfidSocket.readyState === WebSocket.OPEN) {
                scannedTagsCount = 0;
                countSpan.textContent = `(0)`;
                rfidSocket.send(JSON.stringify({ action: "start_scan", mode: 'multiple', datacenter_id: parseInt(datacenterId) }));
                startBtn.disabled = true; stopBtn.disabled = false; cancelBtn.disabled = false; 
                tagsListDiv.innerHTML = '';
            }
        });
        stopBtn.addEventListener('click', () => {
            if (rfidSocket.readyState === WebSocket.OPEN) {
                rfidSocket.send(JSON.stringify({ action: "stop_scan", datacenter_id: parseInt(dcSelect.value) }));
            }
        });
        cancelBtn.addEventListener('click', () => {
            if (rfidSocket.readyState === WebSocket.OPEN) {
                rfidSocket.send(JSON.stringify({ action: "cancel_scan" }));
            }
        });
    }

    // Script para as Abas de Visualização
    const tabButtons = document.querySelectorAll('#inventory-tabs .nav-link');
    const tabPanes = document.querySelectorAll('.tab-content .tab-pane');
    tabButtons.forEach(button => {
        button.addEventListener('click', (event) => {
            tabButtons.forEach(btn => btn.classList.remove('active'));
            tabPanes.forEach(pane => pane.classList.remove('active'));
            event.currentTarget.classList.add('active');
            const targetPaneId = event.currentTarget.getAttribute('data-bs-target');
            document.querySelector(targetPaneId).classList.add('active');
        });
    });
});

                    pontoDiv.style.left = coords.left;
                    pontoDiv.style.transform = 'translate(-50%, -50%)';

                    // --- NOVA LÓGICA DE CORES ---
                    if (quarto.status_embarcado === 'Offline') {
                        pontoDiv.style.backgroundColor = '#6c757d'; // Cinza para Offline
                    } else if (quarto.numero_ativos > 0) {
                        pontoDiv.style.backgroundColor = '#28a745'; // Verde para Ocupado
                    } else {
                        pontoDiv.style.backgroundColor = '#dc3545'; // Vermelho para Livre (mas Online)
                    }

                    const contagemP = document.createElement('p');
                    contagemP.className = 'quarto-asset-count';
                    const textoAtivos = quarto.numero_ativos === 1 ? '1 ativo' : `${quarto.numero_ativos} ativos`;
                    
                    if (quarto.status_embarcado === 'Offline') {
                        contagemP.textContent = 'Embarcado Offline';
