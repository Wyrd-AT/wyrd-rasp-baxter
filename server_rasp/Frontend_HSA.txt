    <script>
    // Roda APENAS UMA VEZ, depois que todo o HTML da página foi carregado.
    document.addEventListener('DOMContentLoaded', function() {
    
        // 1. Ativa todos os ícones da biblioteca Feather na página.

        // 2. Lógica da Sidebar e do Overlay (fundo escurecido).
        const sidebar = document.getElementById('sidebar');
        const toggleButton = document.getElementById('sidebar-toggle');
        const overlay = document.getElementById('overlay');

        if (toggleButton && sidebar && overlay) {
            const toggleSidebar = () => {
                sidebar.classList.toggle('expanded');
                overlay.classList.toggle('active');
            };
            toggleButton.addEventListener('click', toggleSidebar);
            overlay.addEventListener('click', toggleSidebar);
        }

        // 3. Lógica de Interatividade da Tabela (para Ativos e Embarcados).
        const table = document.querySelector('.data-table');
        // Só continua se a página atual TIVER uma tabela com a classe '.data-table'.
        if (table) {
            const editBtn = table.closest('.list-column').querySelector('.toolbar-actions .icon-btn[title="Editar"]');
            const deleteBtn = table.closest('.list-column').querySelector('.toolbar-actions .icon-btn[title="Apagar"]');
            let selectedRow = null;

            table.addEventListener('click', (event) => {
                const row = event.target.closest('.selectable-row');
                if (!row) return;

                if (selectedRow) {
                    selectedRow.style.backgroundColor = '';
                }
                row.style.backgroundColor = '#e0eafc'; // Cor de seleção
                selectedRow = row;
            });

            if (editBtn) {
                editBtn.addEventListener('click', () => {
                    if (!selectedRow) {
                        alert('Por favor, selecione um item na tabela para editar.');
                        return;
                    }
                    const itemId = selectedRow.dataset.id;
                    const pageTitle = document.querySelector('h1').textContent.toLowerCase();
                    let editUrl = '';

                    if (pageTitle.includes('ativos')) {
                        editUrl = `/assets/${itemId}/edit`;
                    } else if (pageTitle.includes('embarcados')) {
                        editUrl = `/embarcados/${itemId}/edit`;
                    }

                    if (editUrl) {
                        window.location.href = editUrl;
                    }
                });
            }

            if (deleteBtn) {
                deleteBtn.addEventListener('click', () => {
                    if (!selectedRow) {
                        alert('Por favor, selecione um item na tabela para apagar.');
                        return;
                    }
                    if (confirm('Tem certeza que deseja apagar o item selecionado?')) {
                        const itemId = selectedRow.dataset.id;
                        const pageTitle = document.querySelector('h1').textContent.toLowerCase();
                        let deleteUrl = '';

                        if (pageTitle.includes('ativos')) {
                            deleteUrl = `/assets/${itemId}/delete`; 
                        } else if (pageTitle.includes('embarcados')) {
                            deleteUrl = `/embarcados/${itemId}/delete`;
                        }

                        if (deleteUrl) {
                            window.location.href = deleteUrl;
                        }
                    }
                });
            }
        }
        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }
                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });
    </script>

<script>
document.addEventListener('DOMContentLoaded', function () {
    // --- LÓGICA DO MODAL (sem alterações) ---
    const modal = document.getElementById('rssi-modal');
    const modalTitle = document.getElementById('modal-title');
    const modalBody = document.getElementById('modal-body-content');
    const closeBtn = document.getElementById('modal-close-btn');

    if (modal && closeBtn) {
        closeBtn.onclick = () => { modal.style.display = "none"; };
        window.onclick = (event) => {
            if (event.target == modal) {
                modal.style.display = "none";
            }
        };
    }

    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation();
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });
            this.nextElementSibling.classList.toggle('show');
        });
    });

    document.querySelectorAll('.test-rssi-btn').forEach(button => {
        button.addEventListener('click', function() {
            if (!myClientId) {
                alert("Ainda a conectar ao servidor, por favor tente novamente em segundos.");
                return;
            }
            const embarcadoId = this.dataset.embarcadoId;

            fetch("{{ url_for('test_rssi_esp') }}", {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({
                    embarcado_id: parseInt(embarcadoId),
                    client_id: myClientId // Envia o nosso "crachá"
                })
            });
        });
    });

    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });
    
    const rssiThresholds = JSON.parse('{{ rssi_thresholds | safe }}');
    let assetNameMap = {};

    fetch("{{ url_for('get_assets_map') }}")
        .then(response => response.json())
        .then(data => { assetNameMap = data; });

    // 2. "Ouvinte" do WebSocket
    window.addEventListener('websocket_message', function (event) {
        try {
            const data = JSON.parse(event.detail); 
            
            if (data.type === "RSSI_REPORT") {
                const modalTitle = document.getElementById('modal-title');
                const modalBody = document.getElementById('modal-body-content');
                const esp_id = data.esp_id;

                // 3. Determina qual limite usar para esta ESP
                const threshold = rssiThresholds.individuais[esp_id] || rssiThresholds.global;
                
                modalTitle.textContent = `Relatório de RSSI da ESP: ${esp_id} (${threshold} dBm)`;
                
                let reportHtml = `
                    <table class="data-table">
                        <thead>
                            <tr>
                                <th>Nome do Ativo</th>
                                <th>MAC Address</th>
                                <th>RSSI</th>
                            </tr>
                        </thead>
                        <tbody>
                `;

                if (data.report && data.report.length > 0) {
                    data.report.sort((a, b) => b.rssi - a.rssi);
                    
                    data.report.forEach(beacon => {
                        const assetName = assetNameMap[beacon.mac.toLowerCase()] || '<i>Desconhecido</i>';
                        const rssiValue = beacon.rssi;

                        // 4. Lógica de coloração
                        const rssiClass = rssiValue >= threshold ? 'rssi-strong' : 'rssi-weak';
                        
                        reportHtml += `
                            <tr>
                                <td>${assetName}</td>
                                <td>${beacon.mac}</td>
                                <td class="${rssiClass}">${rssiValue} dBm</td>
                            </tr>
                        `;
                    });
                } else {
                    reportHtml += '<tr><td colspan="3" style="text-align:center;">Nenhum beacon conhecido encontrado.</td></tr>';
                }

                reportHtml += '</tbody></table>';
                
                modalBody.innerHTML = reportHtml;
                document.getElementById('rssi-modal').style.display = "block";
            }
        } catch (e) { /* Ignora outras mensagens */ }
    });
});
</script>

<script>
document.addEventListener('DOMContentLoaded', function () {
    const gridContainer = document.getElementById('planta-grid');

    const coordenadasDosPontos = {
        'quarto-1': { top: '40%', left: '50%' },
        'quarto-2': { top: '50%', left: '70%' },
        'quarto-3': { top: '50%', left: '50%' },
        'quarto-4': { top: '50%', left: '50%' },
        'quarto-5': { top: '65%', left: '50%' },
        'quarto-6': { top: '50%', left: '48%' }
    };

    function atualizarPlanta() {
        fetch("{{ url_for('get_planta_dados') }}")
            .then(response => response.json())
            .then(dadosQuartos => {
                gridContainer.innerHTML = '';

                dadosQuartos.forEach(quarto => {
                    const quartoContainer = document.createElement('div');
                    quartoContainer.className = 'quarto-container';
                    quartoContainer.id = quarto.id_quarto;

                    const tituloH3 = document.createElement('h3');
                    tituloH3.className = 'quarto-titulo';
                    tituloH3.textContent = quarto.nome_quarto;

                    const imgSvg = document.createElement('img');
                    const numeroDoQuarto = quarto.id_quarto.split('-')[1];
                    imgSvg.src = `{{ url_for('static', path='maps/') }}${numeroDoQuarto}.svg`;

                    const pontoDiv = document.createElement('div');
                    pontoDiv.className = 'ponto-status-overlay';
                    const coords = coordenadasDosPontos[quarto.id_quarto] || { top: '50%', left: '50%' };
                    pontoDiv.style.top = coords.top;
                    pontoDiv.style.left = coords.left;
                    pontoDiv.style.transform = 'translate(-50%, -50%)';

                    // --- NOVA LÓGICA DE CORES ---
                    if (quarto.status_embarcado === 'Offline') {
                        pontoDiv.style.backgroundColor = '#6c757d'; // Cinza para Offline
                    } else if (quarto.numero_ativos > 0) {
                        pontoDiv.style.backgroundColor = '#28a745'; // Verde para Ocupado
                    } else {
                        pontoDiv.style.backgroundColor = '#dc3545'; // Vermelho para Livre (mas Online)
                    }

                    const contagemP = document.createElement('p');
                    contagemP.className = 'quarto-asset-count';
                    const textoAtivos = quarto.numero_ativos === 1 ? '1 ativo' : `${quarto.numero_ativos} ativos`;
                    
                    if (quarto.status_embarcado === 'Offline') {
                        contagemP.textContent = 'Embarcado Offline';
                    } else {
                        contagemP.textContent = quarto.numero_ativos > 0 ? textoAtivos : 'Livre';
                    }
                    
                    // --- NOVA LÓGICA DE TOOLTIP ---
                    let tooltipText = `${quarto.nome_quarto}`;
                    if (quarto.status_embarcado === 'Offline') {
                        tooltipText += `\n(Embarcado Offline)`;
                    } else if (quarto.numero_ativos > 0) {
                        // Itera sobre a nova lista de ativos detalhados
                        quarto.ativos.forEach(ativo => {
                            tooltipText += `\n- ${ativo.nome} (Entrada: ${ativo.horario_entrada})`;
                        });
                    } else {
                        tooltipText += `\nLivre`;
                    }
                    pontoDiv.title = tooltipText;

                    quartoContainer.appendChild(tituloH3);
                    quartoContainer.appendChild(imgSvg);
                    quartoContainer.appendChild(pontoDiv);
                    quartoContainer.appendChild(contagemP);
                    
                    gridContainer.appendChild(quartoContainer);
                });
            });
    }

    window.addEventListener('websocket_open', function() {
        console.log("Planta Baixa: WebSocket conectado. Carregando dados iniciais.");
        atualizarPlanta();
    });

    // Ouve pelo evento 'atualizar_estado' disparado pelo base.html
    window.addEventListener('websocket_message', function (event) {
        if (event.detail === "ATUALIZAR_ESTADO") {
            console.log("Planta Baixa: Sinal de atualização recebido.");
            atualizarPlanta();
        }
    });
});
</script>
{% endblock %}
