        const formsContainer = document.getElementById('forms-container');

        // Só executa se o botão e o container existirem nesta página
        if (toggleFormsBtn && formsContainer) {
            
            // Função para atualizar o texto do botão
            const updateButtonText = () => {
                if (formsContainer.classList.contains('collapsed')) {
                    toggleFormsBtn.textContent = 'Mostrar Formulários';
                } else {
                    toggleFormsBtn.textContent = 'Recolher Formulários';
                }
            };

            // Ao carregar a página, verifica se o estado "recolhido" estava salvo
            if (localStorage.getItem('formsCollapsed') === 'true') {
                formsContainer.classList.add('collapsed');
            }
            updateButtonText(); // Garante que o texto esteja correto no carregamento

            // Adiciona o evento de clique ao botão
            toggleFormsBtn.addEventListener('click', () => {
                // Alterna a classe 'collapsed' no container
                formsContainer.classList.toggle('collapsed');
                
                // Atualiza o texto do botão para o novo estado
                updateButtonText();

                // Salva o estado atual no armazenamento local
                localStorage.setItem('formsCollapsed', formsContainer.classList.contains('collapsed'));
            });
        }
        
        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }

                if (event.data === "ATUALIZAR_ESTADO") {
                    console.log("Recebido sinal para atualizar. Recarregando a página...");
                    // Adiciona um pequeno delay para garantir que o banco de dados foi atualizado
                    setTimeout(() => {
                        window.location.reload();
                    }, 500); // 500ms de delay
                }   

                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });
    
    </script>

                    data.report.sort((a, b) => (b.avg_rssi||-1000) - (a.avg_rssi||-1000));
                    data.report.forEach(b => {
                        const name = assetNameMap[b.mac.toLowerCase()] || 'Unknown';

<script>
document.addEventListener('DOMContentLoaded', function () {
    const rendererContainer = document.getElementById('renderer-container');
    const slugPainel = "{{ painel_atual.slug }}";

    function getPriorityStatus(quarto) {
        if (quarto.status_embarcado === 'Offline') return { cor: '#6c757d', animacao: false };
        if (quarto.ativos.some(a => a.status === 'ALERTA')) return { cor: '#dc3545', animacao: true };
        if (quarto.ativos.some(a => a.status === 'PENDENTE')) return { cor: '#ffc107', animacao: false };
        if (quarto.ativos.length > 0) return { cor: '#28a745', animacao: false };
        return { cor: '#007bff', animacao: false }; // Azul (Livre)
    }

    function buildTooltip(quarto) {
        // Tooltip Nativo Simples e Funcional
        let text = `Quarto: ${quarto.nome_quarto}`;
        if (quarto.status_embarcado === 'Offline') {
            text += `\n[!] Sensor Offline`;
        } else if (quarto.ativos.length > 0) {
            text += `\nAtivos:`;
            quarto.ativos.forEach(a => { text += `\n- ${a.nome} (${a.status})`; });
        } else {
            text += `\nStatus: Livre`;
        }
        return text;
    }

    function renderizarPlanta(data) {
        rendererContainer.innerHTML = '';
        
        if (!data.andares || data.andares.length === 0) {
            rendererContainer.innerHTML = '<p style="text-align:center">Nenhum andar configurado.</p>';
            return;
        }

        data.andares.forEach(andar => {
            if (!andar.imagem_url) return;

            const bloco = document.createElement('div');
            bloco.className = 'planta-bloco';
            
            const titulo = document.createElement('h3');
            titulo.textContent = andar.nome_andar;
            bloco.appendChild(titulo);

            const containerDinamico = document.createElement('div');
            containerDinamico.className = 'planta-container-dinamica';

            // 1. Imagem Real
            const img = document.createElement('img');
            img.src = andar.imagem_url;
            img.className = 'img-planta-real';
            img.alt = "Planta";
            img.onerror = function() { console.error("Erro imagem:", andar.imagem_url); };
            
            // 2. Container de Pontos
            const pontosDiv = document.createElement('div');
            pontosDiv.className = 'planta-pontos-container';

            andar.quartos.forEach(quarto => {
                if (quarto.pos_x != null && quarto.pos_y != null) {
                    
                    // Cria o Ponto (Bolinha)
                    const ponto = document.createElement('div');
                    ponto.className = 'ponto-quarto-status';
                    ponto.style.left = `${quarto.pos_x}%`;
                    ponto.style.top = `${quarto.pos_y}%`;
                    
                    const st = getPriorityStatus(quarto);
                    ponto.style.backgroundColor = st.cor;
                    if(st.animacao) ponto.classList.add('blinking-animation');
                    
                    ponto.title = buildTooltip(quarto); // Tooltip nativo

                    // Cria o Rótulo (Texto embaixo)
                    const rotulo = document.createElement('div');
                    rotulo.className = 'ponto-rotulo';
                    rotulo.textContent = quarto.nome_quarto;
                    // Posiciona o rótulo relativo à bolinha
                    ponto.appendChild(rotulo);

                    pontosDiv.appendChild(ponto);
                }
            });

            containerDinamico.appendChild(img);
            containerDinamico.appendChild(pontosDiv);
            bloco.appendChild(containerDinamico);
            rendererContainer.appendChild(bloco);
        });
    }

    function atualizarSumario(sumario) {
        if (!sumario) return;

            .then(data => {
                renderizarPlanta(data);
                atualizarSumario(data.sumario_geral);
            })
            .catch(err => console.error(err));
    }

    window.addEventListener('websocket_message', (e) => {
        if (e.detail === "ATUALIZAR_ESTADO") carregarDados();
    });

    carregarDados();
});

<script>
document.addEventListener('DOMContentLoaded', function () {
    // --- LÓGICA CORRIGIDA DO MENU ---
    const triggers = document.querySelectorAll('.actions-menu-trigger');
    
    triggers.forEach(trigger => {
        trigger.addEventListener('click', function (e) {
            e.stopPropagation();
            const current = this.nextElementSibling;
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== current) d.classList.remove('show');
            });
            current.classList.toggle('show');
        });
    });

    document.addEventListener('click', function (e) {
        if (!e.target.closest('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                d.classList.remove('show');
            });
        }
    });
});
</script>
