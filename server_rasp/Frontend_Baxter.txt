<script>
document.addEventListener('DOMContentLoaded', function () {
    // Lógica para controlar a abertura e fechamento do menu de ações
    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation(); // Impede que o clique feche o menu imediatamente
            
            // Fecha qualquer outro menu que esteja aberto
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });

            // Alterna a exibição do menu clicado
            this.nextElementSibling.classList.toggle('show');
        });
    });

    // Fecha o menu se o usuário clicar em qualquer outro lugar da tela
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });
});
</script>

    // Roda APENAS UMA VEZ, depois que todo o HTML da página foi carregado.
    document.addEventListener('DOMContentLoaded', function() {
    
        // 1. Ativa todos os ícones da biblioteca Feather na página.

        // 2. Lógica da Sidebar e do Overlay (fundo escurecido).
        const sidebar = document.getElementById('sidebar');
        const toggleButton = document.getElementById('sidebar-toggle-btn');
        const overlay = document.getElementById('overlay');

        if (toggleButton && sidebar && overlay) {
            const toggleSidebar = () => {
                sidebar.classList.toggle('expanded');
                overlay.classList.toggle('active');
            };
            toggleButton.addEventListener('click', toggleSidebar);
            overlay.addEventListener('click', toggleSidebar);
        }
        
        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }

                if (event.data === "ATUALIZAR_ESTADO") {
                    console.log("Recebido sinal para atualizar. Recarregando a página...");
                    // Adiciona um pequeno delay para garantir que o banco de dados foi atualizado
                    setTimeout(() => {
                        window.location.reload();
                    }, 500); // 500ms de delay
                }   

                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });
    </script>

document.addEventListener('DOMContentLoaded', function () {

    const pendingSectionContainer = document.getElementById('pending-section-container');
    const pendingBody = document.getElementById('pending-events-tbody');

    function renderPendingEvents(events) {
        if (!events || events.length === 0) {
            pendingSectionContainer.style.display = 'none';
            return;
        }

        pendingSectionContainer.style.display = 'block';
        pendingBody.innerHTML = '';

        // === INÍCIO DA CORREÇÃO ===
        // 1. Criamos um "molde" da URL no servidor, com um placeholder.
        const cancelUrlTemplate = "{{ url_for('cancel_pending_event', asset_mac='PLACEHOLDER') }}";
        // === FIM DA CORREÇÃO ===

        events.forEach(event => {
            const row = document.createElement('tr');
            
            // ... (criação das outras células timeCell, camaCell, etc. continua igual)
            const timeCell = document.createElement('td');
            timeCell.textContent = `${event.tempo_pendente_sec}s`;

            const camaCell = document.createElement('td');
            camaCell.textContent = event.nome_ativo;

            const quartoCell = document.createElement('td');
            quartoCell.textContent = event.quarto_nome;

            const detailsCell = document.createElement('td');
            detailsCell.textContent = event.detalhes;
            
            const actionsCell = document.createElement('td');

            // === INÍCIO DA CORREÇÃO ===
            // 2. Usamos o molde e substituímos o placeholder pelo MAC real do ativo.
            const cancelUrl = cancelUrlTemplate.replace('PLACEHOLDER', event.ativo_mac);
            // === FIM DA CORREÇÃO ===
            
            actionsCell.innerHTML = `
                <form action="${cancelUrl}" method="post" onsubmit="return confirm('Tem certeza que deseja cancelar esta verificação para ${event.nome_ativo}?');">
                    <button type="submit" class="btn-cancel" title="Cancelar verificação em segundo plano">
                        Cancelar
                    </button>
                </form>
            `;

            row.appendChild(timeCell);
            row.appendChild(camaCell);
            row.appendChild(quartoCell);
            row.appendChild(detailsCell);
            row.appendChild(actionsCell);
            
            pendingBody.appendChild(row);
        });
    }

    async function fetchPendingEvents() {
        try {
            const response = await fetch("{{ url_for('get_pending_events') }}");
            if (!response.ok) {
                throw new Error('Falha ao buscar eventos pendentes');
            }
            const events = await response.json();
            renderPendingEvents(events);
        } catch (error) {
            console.error("Erro ao buscar eventos pendentes:", error);
            pendingSectionContainer.style.display = 'none';
        }
    }

    fetchPendingEvents();
    setInterval(fetchPendingEvents, 30000); 
});
