document.addEventListener('DOMContentLoaded', function () {
    // Lógica para controlar a abertura e fechamento do menu de ações
    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation(); // Impede que o clique feche o menu imediatamente
            
            // Fecha qualquer outro menu que esteja aberto
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });

            // Alterna a exibição do menu clicado
            this.nextElementSibling.classList.toggle('show');
        });
    });

    // Fecha o menu se o usuário clicar em qualquer outro lugar da tela
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });
});
</script>

    // Roda APENAS UMA VEZ, depois que todo o HTML da página foi carregado.
    document.addEventListener('DOMContentLoaded', function() {
    
        // 1. Ativa todos os ícones da biblioteca Feather na página.

        // 2. Lógica da Sidebar e do Overlay (fundo escurecido).

        const toggleButton = document.getElementById('sidebar-toggle-btn');
        const overlay = document.getElementById('overlay');

        if (toggleButton && sidebar && overlay) {
            const toggleSidebar = () => {

            };

        }
        
        const toggleFormsBtn = document.getElementById('forms-toggle-btn');
        const formsContainer = document.getElementById('forms-container');

        // Só executa se o botão e o container existirem nesta página
        if (toggleFormsBtn && formsContainer) {
            
            // Função para atualizar o texto do botão
            const updateButtonText = () => {
                if (formsContainer.classList.contains('collapsed')) {
                    toggleFormsBtn.textContent = 'Mostrar Formulários'; 
                } else {
                    toggleFormsBtn.textContent = 'Recolher Formulários'; 
                }
            };

            // Ao carregar a página, verifica se o estado "recolhido" estava salvo
            if (localStorage.getItem('formsCollapsed') === 'true') {
                formsContainer.classList.add('collapsed'); 
            }
            updateButtonText(); // Garante que o texto esteja correto no carregamento

            // Adiciona o evento de clique ao botão
            toggleFormsBtn.addEventListener('click', () => {
                // Alterna a classe 'collapsed' no container
                formsContainer.classList.toggle('collapsed'); 
                
                // Atualiza o texto do botão para o novo estado
                updateButtonText();

                // Salva o estado atual no armazenamento local
                localStorage.setItem('formsCollapsed', formsContainer.classList.contains('collapsed')); 
            });
        }
        
        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }

                if (event.data === "ATUALIZAR_ESTADO") {
                    console.log("Recebido sinal para atualizar. Recarregando a página...");
                    // Adiciona um pequeno delay para garantir que o banco de dados foi atualizado
                    setTimeout(() => {
                        window.location.reload();
                    }, 500); // 500ms de delay
                }   

                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });
    </script>

    const closeBtn = document.getElementById('modal-close-btn');
    if (modal && closeBtn) {
        closeBtn.onclick = () => { modal.style.display = "none"; };
        window.onclick = (event) => {
            if (event.target == modal) {
                modal.style.display = "none";
            }
        };
    }

    // Lógica para controlar o menu de três pontinhos (sem alterações)
    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation();
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });
            this.nextElementSibling.classList.toggle('show');
        });
    });
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });

    // --- INÍCIO DA NOVA LÓGICA ---
    // Mapeamento de MAC para Nome do Ativo (para o relatório)
    const rssiThresholds = JSON.parse('{{ rssi_thresholds | safe }}');
    let assetNameMap = {};
    fetch("{{ url_for('get_assets_map') }}")
        .then(response => response.json())
        .then(data => { assetNameMap = data; });

    // Adiciona o evento de clique para TODOS os botões de "Testar RSSI"
    document.querySelectorAll('.test-rssi-btn').forEach(button => {
        button.addEventListener('click', function() {
            if (!myClientId) {
                alert("Ainda a conectar ao servidor, por favor tente novamente em segundos.");
                return;
            }
            const embarcadoId = this.dataset.embarcadoId;

            // Mostra um feedback visual de carregamento
            const modalTitle = document.getElementById('modal-title');
            const modalBody = document.getElementById('modal-body-content');
            modalTitle.textContent = 'Aguardando Relatório...';
            modalBody.innerHTML = '<p style="text-align:center;">Solicitando dados do servidor...</p>';
            modal.style.display = 'block';

            // Envia o pedido para o backend
            fetch("{{ url_for('test_rssi_esp') }}", {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({
                    embarcado_id: parseInt(embarcadoId),
                    client_id: myClientId // Envia o nosso "crachá" de conexão
                })
            });
        });
    });

    // "Ouvinte" global de mensagens WebSocket, que vai receber o relatório de volta
    window.addEventListener('websocket_message', function (event) {
        try {
            const data = JSON.parse(event.detail); 
            
            if (data.type === "RSSI_REPORT") {
                const modalTitle = document.getElementById('modal-title');
                const modalBody = document.getElementById('modal-body-content');
                const esp_id = data.esp_id;

                const threshold = rssiThresholds.individuais[esp_id] || rssiThresholds.global;
                
                modalTitle.textContent = `Relatório de RSSI da ESP: ${esp_id} (Limite: ${threshold} dBm)`;
                
                let reportHtml = `<table class="data-table"><thead><tr><th>Nome do Ativo</th><th>MAC Address</th><th>RSSI</th></tr></thead><tbody>`;

                if (data.report && data.report.length > 0) {
                    data.report.sort((a, b) => b.rssi - a.rssi); // Ordena do mais forte para o mais fraco
                    data.report.forEach(beacon => {
                        const assetName = assetNameMap[beacon.mac.toLowerCase()] || '<i>Desconhecido</i>';
                        const rssiValue = beacon.rssi;
                        const rssiClass = rssiValue >= threshold ? 'rssi-strong' : 'rssi-weak';
                        
                        reportHtml += `<tr><td>${assetName}</td><td>${beacon.mac}</td><td class="${rssiClass}">${rssiValue} dBm</td></tr>`;
                    });
                } else {
                    reportHtml += '<tr><td colspan="3" style="text-align:center;">Nenhum beacon conhecido encontrado.</td></tr>';
                }

                reportHtml += '</tbody></table>';
                modalBody.innerHTML = reportHtml;
                modal.style.display = "block";
            }
        } catch (e) { /* Ignora outras mensagens que não sejam JSON */ }
    });
    // --- FIM DA NOVA LÓGICA ---
});

    // NOVO: Pegamos o elemento do contador que criamos no HTML
    const countSpan = document.getElementById('scanned-tags-count');
    // NOVO: Criamos uma variável para guardar a contagem
    let scannedTagsCount = 0;

    function resetScanUI() {
        statusDiv.textContent = "Status: Conectado e pronto.";
        tagsListDiv.innerHTML = '<div class="muted">Aguardando início da leitura...</div>';
        scannedTagsCount = 0;
        countSpan.textContent = `(0)`;
        startBtn.disabled = false;
        stopBtn.disabled = true;
        cancelBtn.disabled = true;
    }


    if (startBtn) { 
        const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
        const rfidSocket = new WebSocket(`${ws_protocol}//${window.location.host}/ws/rfid`);
        
        rfidSocket.onopen = () => { resetScanUI(); };
        rfidSocket.onclose = () => { 
            statusDiv.textContent = "Status: Desconectado. Recarregue a página."; 
            startBtn.disabled = true; 
            stopBtn.disabled = true; 
            cancelBtn.disabled = true;
        };
        
        rfidSocket.onmessage = (event) => {
            const data = JSON.parse(event.data);
            
            // NOVO: Bloco para tratar o erro de hardware de forma especial
            if (data.type === 'hardware_error') {
                statusDiv.textContent = data.message;
                statusDiv.style.color = '#dc3545'; // Deixa o texto vermelho
                statusDiv.style.fontWeight = 'bold';
                // Desabilita todos os botões permanentemente até a página ser recarregada
                startBtn.disabled = true;
                stopBtn.disabled = true;
                cancelBtn.disabled = true;
                return; // Interrompe o processamento para não executar outras lógicas
            }
            
            if (data.type === 'scan_status' || data.type === 'scan_error') {
                statusDiv.textContent = `Status: ${data.message}`;
                if (data.message.toLowerCase().includes("finalizada") || data.message.toLowerCase().includes("erro") || data.message.toLowerCase().includes("cancelada")) {
                    resetScanUI();
                    statusDiv.textContent = "Status: Conectado e pronto.";
                }
            }
            if (data.type === 'tag_scanned') {
                if (tagsListDiv.querySelector('.muted')) {
                    tagsListDiv.innerHTML = '';
                }
                const tagElement = document.createElement('code');
                tagElement.textContent = data.tag_id;
                tagsListDiv.appendChild(tagElement);

                // NOVO: Incrementamos o contador e atualizamos o texto na tela
                scannedTagsCount++;
                countSpan.textContent = `(${scannedTagsCount})`;
            }
        };

        startBtn.addEventListener('click', () => {
            const datacenterId = dcSelect.value;
            if (!datacenterId) {
                alert("Por favor, selecione um datacenter primeiro.");
                return;
            }
            if (rfidSocket.readyState === WebSocket.OPEN) {
                // NOVO: Zeramos o contador ao iniciar uma nova leitura
                scannedTagsCount = 0;
                countSpan.textContent = `(0)`;
                
                rfidSocket.send(JSON.stringify({
                    action: "start_scan",
                    datacenter_id: parseInt(datacenterId)
                }));
                statusDiv.textContent = "Status: Iniciando...";
                startBtn.disabled = true;
                stopBtn.disabled = false;
                cancelBtn.disabled = false; 
                tagsListDiv.innerHTML = '<div class="muted">Aguardando confirmação do scan...</div>';
            } else {
                alert("Erro: Conexão com o servidor de scan não está aberta.");
            }
        });

        stopBtn.addEventListener('click', () => {
            const datacenterId = dcSelect.value;
            if (rfidSocket.readyState === WebSocket.OPEN) {
                rfidSocket.send(JSON.stringify({
                    action: "stop_scan",
                    datacenter_id: parseInt(datacenterId)
                }));
            }
        });

        cancelBtn.addEventListener('click', () => {
            if (rfidSocket.readyState === WebSocket.OPEN) {
                rfidSocket.send(JSON.stringify({ action: "cancel_scan" }));
            }
        });
    }
});
