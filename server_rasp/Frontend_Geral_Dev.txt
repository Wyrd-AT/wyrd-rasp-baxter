<script>
document.addEventListener('DOMContentLoaded', function () {
    // Lógica para controlar a abertura e fechamento do menu de ações
    document.querySelectorAll('.actions-menu-trigger').forEach(button => {
        button.addEventListener('click', function (event) {
            event.stopPropagation(); // Impede que o clique feche o menu imediatamente
            
            // Fecha qualquer outro menu que esteja aberto
            document.querySelectorAll('.actions-dropdown.show').forEach(d => {
                if (d !== this.nextElementSibling) d.classList.remove('show');
            });

            // Alterna a exibição do menu clicado
            this.nextElementSibling.classList.toggle('show');
        });
    });

    // Fecha o menu se o usuário clicar em qualquer outro lugar da tela
    window.addEventListener('click', (event) => {
        if (!event.target.matches('.actions-menu-trigger')) {
            document.querySelectorAll('.actions-dropdown.show').forEach(d => d.classList.remove('show'));
        }
    });
});
</script>
{% endblock %}

    // Roda APENAS UMA VEZ, depois que todo o HTML da página foi carregado.
    document.addEventListener('DOMContentLoaded', function() {
    
        // 1. Ativa todos os ícones da biblioteca Feather na página.

        // 2. Lógica da Sidebar e do Overlay (fundo escurecido).
        const sidebar = document.getElementById('sidebar');
        const toggleButton = document.getElementById('sidebar-toggle-btn');
        const overlay = document.getElementById('overlay');

        if (toggleButton && sidebar && overlay) {
            const toggleSidebar = () => {
                sidebar.classList.toggle('expanded');
                overlay.classList.toggle('active');
            };
            toggleButton.addEventListener('click', toggleSidebar);
            overlay.addEventListener('click', toggleSidebar);
        }
        
        const toggleFormsBtn = document.getElementById('forms-toggle-btn');
        const formsContainer = document.getElementById('forms-container');

        // Só executa se o botão e o container existirem nesta página
        if (toggleFormsBtn && formsContainer) {
            
            // Função para atualizar o texto do botão
            const updateButtonText = () => {
                if (formsContainer.classList.contains('collapsed')) {
                    toggleFormsBtn.textContent = 'Mostrar Formulários';
                } else {
                    toggleFormsBtn.textContent = 'Recolher Formulários';
                }
            };
            
            // Ao carregar a página, verifica se o estado "recolhido" estava salvo
            if (localStorage.getItem('formsCollapsed') === 'true') {
                formsContainer.classList.add('collapsed');
            }
            updateButtonText(); // Garante que o texto esteja correto no carregamento

            // Adiciona o evento de clique ao botão
            toggleFormsBtn.addEventListener('click', () => {
                // Alterna a classe 'collapsed' no container
                formsContainer.classList.toggle('collapsed');
                
                // Atualiza o texto do botão para o novo estado
                updateButtonText();

                // Salva o estado atual no armazenamento local
                localStorage.setItem('formsCollapsed', formsContainer.classList.contains('collapsed'));
            });
        }

        
        function conectarWebSocket() {
            const ws_protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
            const socket = new WebSocket(`${ws_protocol}//${window.location.host}/ws`);

            socket.onopen = function(event) {
                console.log("SUCESSO: Conexão WebSocket global estabelecida.");
                // Anuncia para toda a aplicação que a conexão está aberta
                window.dispatchEvent(new Event('websocket_open'));
            };

            socket.onmessage = function(event) {
                // Ignora os pings para não poluir a consola
                if (event.data === "ping") {
                    return;
                }

                try {
                    const data = JSON.parse(event.data);
                    // Verifica se é a mensagem especial de conexão para receber o nosso "crachá"
                    if (data.type === "CONNECTION_INFO") {
                        myClientId = data.client_id;
                        console.log("Crachá de cliente recebido:", myClientId);
                        return; // Não precisa de anunciar esta mensagem para outras páginas
                    }
                } catch (e) {
                    // A mensagem não é um JSON, provavelmente é "ATUALIZAR_ESTADO"
                }
                
                // Anuncia TODAS as outras mensagens para que as páginas específicas possam ouvi-las
                window.dispatchEvent(new CustomEvent('websocket_message', { detail: event.data }));
            };

            socket.onclose = function(event) {
                console.warn("AVISO: Conexão WebSocket perdida. Tentando reconectar em 5s...");
                setTimeout(conectarWebSocket, 5000);
            };

            socket.onerror = function(error) {
                console.error("ERRO: Erro na conexão WebSocket:", error);
                socket.close(); // Força o onclose para acionar a reconexão
            };
        }
        conectarWebSocket();
    });
    </script>

                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({

                    client_id: myClientId // Envia o nosso "crachá" de conexão

  // --- INÍCIO DA SEÇÃO NOVA ---
  // 2. Adiciona um "ouvinte" para o evento de SUBMIT do formulário
  inventoryForm.addEventListener('submit', function(event) {
    // Se o buffer (nossa lista de itens) estiver vazio...
    if (buffer.length === 0) {
      // Impede o envio do formulário
      event.preventDefault();
      // Mostra um alerta para o usuário
      alert("Por favor, adicione pelo menos um item ao inventário antes de salvar.");
    }
    // Se o buffer não estiver vazio, o formulário é enviado normalmente.
  });
  // --- FIM DA SEÇÃO NOVA ---
  
  function renderPendingItems(){
    if(buffer.length === 0){
      pendentesDiv.innerHTML = '<div class="muted">Nenhum item adicionado para salvar.</div>';
      hiddenInputsDiv.innerHTML = '';
      return;
    }
    pendentesDiv.innerHTML = '<ul>' + buffer.map((item, index) => (
      `<li style="display:flex; align-items:center; gap:6px; margin-bottom: 4px;">
        <code>${item.codigo}</code> — ${item.tipoNome}
        <button type="button" class="icon-btn" title="Remover" onclick="removePendingItem(${index})"
                style="display:inline-flex; align-items:center; justify-content:center;
                       width:20px; height:20px; padding:0; margin:0; border-radius: 50%;">
          <span class="svg-icon" style="--icon-url: url({{ url_for('static', path='icons/close.svg') }}); width:14px; height:14px;"></span>
        </button>
      </li>`
    )).join('') + '</ul>';
    hiddenInputsDiv.innerHTML = buffer.map(item =>
      `<input type="hidden" name="codigos" value="${item.codigo}"/>
       <input type="hidden" name="tipos" value="${item.tipoId}"/>`
    ).join('');
  }
  window.removePendingItem = (index) => {
    buffer.splice(index, 1);
    renderPendingItems();
  }
  addBtn?.addEventListener('click', () => {
    const codigo = (rfidInput.value || '').trim().toUpperCase();
    if(!codigo){ rfidInput.focus(); return; }
    if (buffer.some(item => item.codigo === codigo)) {
        // Silencioso: só limpa e segue o baile
        rfidInput.value = '';
        // opcional: manter foco para leitura contínua
        rfidInput.focus();
        return;
    }
    const tipoId = tipoSelect.value;
    const tipoNome = tipoSelect.options[tipoSelect.selectedIndex].text;
    buffer.push({codigo: codigo, tipoId: tipoId, tipoNome: tipoNome});
    rfidInput.value = ''; rfidInput.focus(); renderPendingItems();
  });
  rfidInput?.addEventListener('keydown', e => {
    if(e.key === 'Enter'){ e.preventDefault(); addBtn.click(); }
  });
  clearBtn?.addEventListener('click', () => { buffer = []; renderPendingItems(); });
  renderPendingItems();
});
